Action<Mesh> feather=m=>{

Action<Transform,string,Vector2[],float> road=(parent,name,points,width)=>{
    var samples=new System.Collections.Generic.List<Vector2>();
    for(int segment=1;segment<points.Length;segment++){
        var start=points[segment-1];var end=points[segment];
        int n=Mathf.CeilToInt(Vector2.Distance(start,end)/.62f);
        for(int k=0;k<n;k++)samples.Add(Vector2.Lerp(start,end,k/(float)n));
    }
    samples.Add(points.Last());
    var vertices=new System.Collections.Generic.List<Vector3>();
    var tris=new System.Collections.Generic.List<int>();
    for(int i=0;i<samples.Count;i++){
        var p=samples[i];var d=(samples[Mathf.Min(i+1,samples.Count-1)]-samples[Mathf.Max(i-1,0)]).normalized;
        var side=new Vector2(-d.y,d.x);
        float half=width*.5f*(.88f+.10f*Mathf.PerlinNoise(p.x*.10f,p.y*.10f));
        for(int j=0;j<9;j++){
            var q=p+side*half*((j-4)*.25f);
            vertices.Add(new Vector3(q.x,h(q.x,q.y)+.045f,q.y));
            if(i==0||j==0)continue;
            int k=vertices.Count-1;
            tris.AddRange(new[]{k-10,k-1,k-9,k-1,k,k-9});
        }
    }
    var vv=vertices.ToArray();
    if(Vector3.Cross(vv[tris[1]]-vv[tris[0]],vv[tris[2]]-vv[tris[0]]).y<0)
       for(int i=0;i<tris.Count;i+=3){int swap=tris[i+1];tris[i+1]=tris[i+2];tris[i+2]=swap;}
    var obj=meshObject(parent,name,vv,new[]{tris.ToArray()},new[]{trail});
    obj.GetComponent<MeshRenderer>().shadowCastingMode=UnityEngine.Rendering.ShadowCastingMode.Off;
    var mesh=obj.GetComponent<MeshFilter>().sharedMesh;
    feather(mesh);
    Color[] colors=mesh.colors;
    for(int i=0;i<colors.Length;i++){var c=colors[i];c.a*=Mathf.SmoothStep(0,1,Mathf.Min(i/9,samples.Count-1-i/9)/6f);colors[i]=c;}
    mesh.colors=colors;
};

    UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
    return "포구, 생활 소품, 절벽 연결과 전망 마당, 카메라 분류 적용";
}catch{Undo.RevertAllDownToGroup(group);throw;}
