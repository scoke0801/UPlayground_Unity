Func<Vector3[],Vector2,Vector2> closest=(line,p)=>{
    var best=new Vector2(float.PositiveInfinity,0);
    for(int i=1;i<line.Length;i++){
        var a=new Vector2(line[i-1].x,line[i-1].z);
        var dir=new Vector2(line[i].x,line[i].z)-a;
        float t=Mathf.Clamp01(Vector2.Dot(p-a,dir)/dir.sqrMagnitude);
        float d=(p-a-dir*t).sqrMagnitude;
        if(d<best.x) best=new Vector2(d,Mathf.Lerp(line[i-1].y,line[i].y,t));
    }
    best.x=Mathf.Sqrt(best.x);
    return best;
};
