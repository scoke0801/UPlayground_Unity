// 1m 격자 수면: 구역 안에서 모서리 중 하나라도 지형이 수위 아래인 칸만 남긴다(가장자리는 지형 아래로 묻혀 얕은 물 페이드로 사라진다).
Func<string, float, float, float, Func<float, float, float>, Mesh> Grid = (name, sMin, sMax, level, halfAt) =>
{
    var verts = new List<Vector3>(); var uv = new List<Vector2>(); var tris = new List<int>(); var index = new Dictionary<long, int>();
    Func<int, int, int> V = (ix, iz) => { long key = ((long)ix << 32) ^ (uint)iz; int i; if (index.TryGetValue(key, out i)) return i; i = verts.Count; verts.Add(new Vector3(ix, level, iz)); uv.Add(new Vector2(ix / 6f, iz / 6f)); index[key] = i; return i; };
    for (int iz = (int)RegionMin.y; iz < (int)RegionMax.y; iz++)
    for (int ix = (int)RegionMin.x; ix < (int)RegionMax.x; ix++)
    {
        var c = new Vector3(ix + 0.5f, 0f, iz + 0.5f); var sq = ToSQ(c);
        if (Mathf.Abs(sq.y) > halfAt(sq.x, sq.y) + 4f) continue;
        // 네 모서리가 모두 구역 안이어야 한다. 낙구 너머로 걸친 칸은 절벽 위 허공에 수면 판을 남긴다.
        float s0 = ToSQ(new Vector3(ix, 0f, iz)).x, s1 = ToSQ(new Vector3(ix + 1, 0f, iz)).x, s2 = ToSQ(new Vector3(ix, 0f, iz + 1)).x, s3 = ToSQ(new Vector3(ix + 1, 0f, iz + 1)).x;
        if (Mathf.Min(Mathf.Min(s0, s1), Mathf.Min(s2, s3)) < sMin || Mathf.Max(Mathf.Max(s0, s1), Mathf.Max(s2, s3)) > sMax) continue;
        float low = Mathf.Min(Mathf.Min(Ground(ix, iz), Ground(ix + 1, iz)), Mathf.Min(Ground(ix, iz + 1), Ground(ix + 1, iz + 1)));
        if (low > level - 0.02f) continue;
        int a = V(ix, iz), b = V(ix + 1, iz), d = V(ix, iz + 1), e = V(ix + 1, iz + 1);
        tris.AddRange(new[] { a, d, b, b, d, e });
    }
    var mesh = new Mesh { name = name }; mesh.SetVertices(verts); mesh.SetUVs(0, uv); mesh.SetTriangles(tris, 0); mesh.RecalculateNormals(); mesh.RecalculateTangents(); mesh.RecalculateBounds();
    return mesh;
};
// 낙구 접근부는 상류 수면을 8mm 위에서 따라가다 중력 곡선으로 떨어진다. 양면 두께를 가진 낙수 체적(호수 폭포 R3 방식).
Func<string, float, float, float, float, float, float, float, Mesh> Fall = (name, lipS, approach, lipY, landY, half0, half1, throwLength) =>
{
    var v = new List<Vector3>(); var uv = new List<Vector2>(); var color = new List<Color>(); var triangles = new List<int>();
    const int Rows = 90, Cols = 48; int skin = (Rows + 1) * (Cols + 1); float drop = lipY - landY;
    for (int side = 0; side < 2; side++)
    for (int row = 0; row <= Rows; row++)
    {
        float a = row / (float)Rows, s, y, flow, transition, thickness; Vector3 tangent;
        if (a < 0.22f) { float t = a / 0.22f; s = Mathf.Lerp(lipS - approach, lipS, t); y = lipY + 0.008f - 0.1f * Mathf.SmoothStep(0f, 1f, Mathf.InverseLerp(0.6f, 1f, t)); flow = t * 0.16f; transition = Mathf.SmoothStep(0f, 1f, Mathf.InverseLerp(0f, 0.68f, t)); thickness = 0f; tangent = F; }
        else { float k = (a - 0.22f) / 0.78f; s = lipS + throwLength * k; y = lipY - 0.092f - (drop * 0.1f) * k - (drop * 0.9f - 0.092f + 0.05f) * k * k; flow = Mathf.Lerp(0.16f, 1f, k); transition = 1f; thickness = 0.14f * Mathf.Sin(k * Mathf.PI); tangent = (F * throwLength - Vector3.up * (drop * 0.1f + 2f * drop * 0.9f * k)).normalized; }
        var normal = Vector3.Cross(tangent, R).normalized; float half = Mathf.Lerp(half0, half1, Mathf.InverseLerp(0.22f, 1f, a));
        for (int col = 0; col <= Cols; col++)
        {
            float u = col / (float)Cols, k = Mathf.InverseLerp(0.22f, 1f, a);
            float ripple = (Mathf.PerlinNoise(u * 6.3f + 3f, k * 4.1f) - 0.5f) * 0.22f * k;
            var p = O + F * (s + ripple) + R * ((u * 2f - 1f) * half); p.y = y + (Mathf.PerlinNoise(u * 5.4f, 4f) - 0.5f) * 0.24f * Mathf.Sin(a * Mathf.PI);
            p += normal * thickness * (side == 0 ? 0.5f : -0.5f);
            v.Add(p); uv.Add(new Vector2(Mathf.Lerp(0.015f, 0.985f, u), flow)); color.Add(new Color(transition, 0f, 0f, 1f));
        }
    }
    for (int side = 0; side < 2; side++) for (int row = 0; row < Rows; row++) for (int col = 0; col < Cols; col++)
    { int a = side * skin + row * (Cols + 1) + col, b = a + Cols + 1; if (side == 0) triangles.AddRange(new[] { a, b, a + 1, a + 1, b, b + 1 }); else triangles.AddRange(new[] { a, a + 1, b, a + 1, b + 1, b }); }
    foreach (int col in new[] { 0, Cols }) for (int row = 0; row < Rows; row++) { int a = row * (Cols + 1) + col, b = a + Cols + 1; triangles.AddRange(new[] { a, a + skin, b, b, a + skin, b + skin }); }
    var mesh = new Mesh { name = name }; mesh.SetVertices(v); mesh.SetUVs(0, uv); mesh.SetColors(color); mesh.SetTriangles(triangles, 0); mesh.RecalculateNormals(); mesh.RecalculateTangents(); mesh.RecalculateBounds();
    return mesh;
};
