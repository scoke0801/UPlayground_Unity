foreach (var n in new[] { "암반", "식생" }) { var old = root.transform.Find(n); if (old != null) Undo.DestroyObjectImmediate(old.gameObject); }
Func<string, object[][], int> Place = (groupName, list) =>
{
    var group = new GameObject(groupName); Undo.RegisterCreatedObjectUndo(group, "북부 폭포 꾸미기"); group.transform.SetParent(root.transform, false);
    int count = 0;
    foreach (var e in list)
    {
        // 구매 팩 하위 폴더 구성이 달라도 같은 이름의 원본을 찾는다. 이름이 겹치면 모호성 오류로 멈춘다.
        string path = (string)e[0];
        if (AssetDatabase.LoadAssetAtPath<GameObject>(path) == null)
        {
            string file = System.IO.Path.GetFileName(path);
            var hits = AssetDatabase.FindAssets(System.IO.Path.GetFileNameWithoutExtension(file) + " t:Prefab", new[] { Stones.TrimEnd('/') }).Select(AssetDatabase.GUIDToAssetPath).Where(p => System.IO.Path.GetFileName(p) == file).ToArray();
            if (hits.Length != 1) throw new Exception("프리팹 경로 " + file + " 후보 " + hits.Length + "개");
            path = hits[0];
        }
        var prefab = AssetDatabase.LoadAssetAtPath<GameObject>(path);
        float s = (float)e[1], q = (float)e[2], size = (float)e[3], yaw = (float)e[4], bury = (float)e[5];
        var go = (GameObject)PrefabUtility.InstantiatePrefab(prefab, group.transform);
        var pos = At(s, q); pos.y = tile.SampleHeight(pos) + ty;
        go.transform.SetPositionAndRotation(pos, Quaternion.LookRotation(F) * Quaternion.Euler(0f, yaw, 0f));
        var renderers = go.GetComponentsInChildren<Renderer>(); var b = renderers[0].bounds; foreach (var r in renderers) b.Encapsulate(r.bounds);
        if (size > 0f) { go.transform.localScale *= size / Mathf.Max(b.size.x, b.size.z); b = renderers[0].bounds; foreach (var r in renderers) b.Encapsulate(r.bounds); }
        // 경사면에서는 가장 낮은 접지점 기준으로 묻어 한쪽이 뜨지 않게 한다.
        float low = Mathf.Min(Mathf.Min(tile.SampleHeight(new Vector3(b.min.x, 0f, b.min.z)), tile.SampleHeight(new Vector3(b.max.x, 0f, b.min.z))), Mathf.Min(tile.SampleHeight(new Vector3(b.min.x, 0f, b.max.z)), tile.SampleHeight(new Vector3(b.max.x, 0f, b.max.z)))) + ty;
        float ground = Mathf.Min(low, pos.y);
        if (e.Length > 7) { float stretch = Mathf.Clamp(((float)e[6] - (float)e[7]) / b.size.y, 1f, 2.2f); var sc = go.transform.localScale; go.transform.localScale = new Vector3(sc.x, sc.y * stretch, sc.z); b = renderers[0].bounds; foreach (var r in renderers) b.Encapsulate(r.bounds); }
        if (e.Length > 6) { float top = (float)e[6]; if (top < 0f) top = pos.y + b.size.y + top; go.transform.position += Vector3.up * (top - b.max.y); }
        else go.transform.position += Vector3.up * (ground - b.min.y - b.size.y * bury);
        go.name = prefab.name + "_" + (count + 1); count++;
    }
    return count;
};
