Action<Vector2[],bool> buildRibbon = (anchors, main) =>
{
    var dense = new System.Collections.Generic.List<Vector2> { anchors[0] };
    for (int i = 1; i < anchors.Length - 1; i++)
    {
        Vector2 p = anchors[i], incoming = anchors[i - 1] - p, outgoing = anchors[i + 1] - p;
        float trim = Mathf.Min(main ? 18f : 12f, .48f * Mathf.Min(incoming.magnitude, outgoing.magnitude));
        Vector2 a = p + incoming.normalized * trim, b = p + outgoing.normalized * trim;
        dense.Add(a);
        for (int j = 1; j <= 32; j++)
        {
            float u = j / 32f;
            dense.Add(Vector2.Lerp(Vector2.Lerp(a, p, u), Vector2.Lerp(p, b, u), u));
        }
    }
    dense.Add(anchors.Last());
    var arc = new float[dense.Count];
    for (int i = 1; i < arc.Length; i++) arc[i] = arc[i - 1] + Vector2.Distance(dense[i - 1], dense[i]);
    float length = arc.Last();
    int count = Mathf.CeilToInt(length / spacing) + 1;
    var path = new Vector2[count];
    for (int i = 0, k = 1; i < count; i++)
    {
        float d = length * i / (count - 1);
        while (k < arc.Length - 1 && arc[k] < d) k++;
        path[i] = Vector2.Lerp(dense[k-1], dense[k], Mathf.InverseLerp(arc[k-1], arc[k], d));
    }
    pathCenters.Add(path);

    const int across = 24, columns = across + 1;
    float shoulder = main ? 4.0f : 3.8f;
    var vertices = new Vector3[count * columns];
    var normals = new Vector3[vertices.Length];
    var colors = new Color32[vertices.Length];
    var faces = new int[(count - 1) * across * 6];
    float radius = float.PositiveInfinity, grade = 0f, cross = 0f;
    var at = new Vector2[3];
    var blockers = new System.Collections.Generic.HashSet<string>();
    var colliders = new Collider[32];
    for (int i = 0; i < count; i++)
    {
        Vector2 p = path[i], a = path[Mathf.Max(0, i - 1)], b = path[Mathf.Min(count - 1, i + 1)];
        Vector2 tangent = (b - a).normalized, side = new Vector2(-tangent.y, tangent.x);
        float d = length * i / (count - 1);
        float envelope = main ? smoother(Mathf.Min(d, length - d) / 14f) :
            Mathf.Lerp(.18f, 1f, smoother(d / 4f)) * smoother((length - d) / 8f);
        float slow = Mathf.PerlinNoise(p.x * .046f, p.y * .046f);
        float width = (main ? 2.09f : 1.59f) * Mathf.Lerp(.91f, 1.15f, slow);
        if (main) width += .35f * (1f - smoother(Vector2.Distance(p, coastAnchors[0]) / 8f));
        else width *= Mathf.Lerp(1.27f, 1f, smoother(d / 9f));
        if (i > 0 && i < count - 1)
        {
            float angle = Vector2.Angle(p - a, b - p) * Mathf.Deg2Rad;
            if (angle > .0001f)
            {
                float current = (Vector2.Distance(p, a) + Vector2.Distance(b, p)) * .5f / angle;
                if (current < radius) { radius = current; at[0] = p; }
            }
        }
        for (int j = 0; j < columns; j++)
        {
            int v = i * columns + j;
            float offset = (j * 2f / across - 1f) * shoulder;
            var q = p + side * offset;
            float u = (q.x - origin.x) / size.x, w = (q.y - origin.z) / size.z;
            if (u <= 0f || u >= 1f || w <= 0f || w >= 1f || ground.IsHole((int)(u * ground.holesResolution),(int)(w * ground.holesResolution)))
                throw new InvalidOperationException("지면 누락 " + q);
            float y = ground.GetInterpolatedHeight(u, w) + origin.y;
            var foot = new Vector3(q.x, y, q.y);
            vertices[v] = road.transform.InverseTransformPoint(foot + Vector3.up * .033f);
            normals[v] = road.transform.InverseTransformDirection(ground.GetInterpolatedNormal(u, w));
            float wind = Mathf.PerlinNoise(q.x * .14f + q.y * .11f, q.y * .20f);
            float grit = Mathf.PerlinNoise(q.x * .91f + 6f, q.y * .89f);
            float sideDrift = (Mathf.PerlinNoise(p.x * .102f, p.y * .122f) - .5f) * .88f;
            float signed = offset + sideDrift;
            float lateral = Mathf.Abs(signed);
            float blow = Mathf.PerlinNoise(q.x * .168f - q.y * .21f + 3f, q.y * .081f);
            float edgeVariation = (wind - .5f) * .37f + (grit - .5f) * .20f;
            float coreWear = 1f - smoother((lateral - width * .47f + edgeVariation) / (width * .72f));
            float shoulderWear = 1f - smoother((lateral - width + (wind - .5f) * 1.00f) / 1.02f);
            float cut = Mathf.Lerp(.35f, 1f, smoother((blow - .26f) / .34f));
            float used = main
                ? 1f - smoother(Mathf.Abs(signed - width * Mathf.Lerp(-.47f, .44f, slow)) / 1.05f)
                : 1f - smoother(lateral / 1.31f);
            float deposited = (1f - smoother(Mathf.Abs(lateral - width * Mathf.Lerp(.67f, 1.38f, wind)) / 1.34f))
                * smoother((wind * .56f + grit * .44f - .34f) / .46f);
            float worn = Mathf.Clamp01(coreWear * (.71f + .16f * used + .14f * grit) + shoulderWear * .14f * cut);
            float edgeCover = shoulderWear * (.16f + .19f * cut) + deposited * .13f;
            float coverage = envelope * Mathf.Lerp(edgeCover, Mathf.Lerp(.61f, .73f, grit), coreWear);
            if (j == 0 || j == across) coverage = 0f;
            colors[v] = (Color32)new Color(worn, deposited, Mathf.Lerp(.27f, .82f, grit), Mathf.Clamp01(coverage));

            if (i > 0)
            {
                var delta = vertices[v] - vertices[v - columns];
                float angle = Mathf.Atan2(Mathf.Abs(delta.y), new Vector2(delta.x, delta.z).magnitude) * Mathf.Rad2Deg;
                if (angle > grade) { grade = angle; at[1] = q; }
            }
            if (j > 0)
            {
                float angle = Mathf.Atan2(Mathf.Abs(vertices[v].y - vertices[v-1].y), 2f * shoulder / across) * Mathf.Rad2Deg;
                if (angle > cross) { cross = angle; at[2] = q; }
            }
            int overlaps = Physics.OverlapCapsuleNonAlloc(foot + Vector3.up * .45f, foot + Vector3.up * 1.5f, .33f, colliders, ~0, QueryTriggerInteraction.Ignore);
            if (overlaps == colliders.Length) throw new InvalidOperationException("충돌 검사 버퍼 초과 " + q);
            for (int hit = 0; hit < overlaps; hit++)
                if (!(colliders[hit] is TerrainCollider)) blockers.Add(colliders[hit].name + " @ " + q);
            if (i == 0 || j == 0) continue;
            int f = ((i - 1) * across + j - 1) * 6;
            faces[f] = v - columns - 1; faces[f + 1] = v - columns; faces[f + 2] = v - 1;
            faces[f + 3] = v - 1; faces[f + 4] = v - columns; faces[f + 5] = v;
        }
    }
    if (Vector3.Dot(Vector3.Cross(vertices[faces[1]] - vertices[faces[0]], vertices[faces[2]] - vertices[faces[0]]), normals[faces[0]]) < 0)
        for (int i = 0; i < faces.Length; i += 3)
        { int swap = faces[i+1]; faces[i+1] = faces[i+2]; faces[i+2] = swap; }
    string id = main ? "R5-junction" : "R5-lookout-coast";
    if (radius < minimumRadius || grade > maximumGrade || cross > maximumGrade || blockers.Count > 0)
        errors.Add(id + ": " + string.Join("; ", blockers.Take(12)));
    string[] fields = { "curvatureAt", "longitudinalAt", "crossAt" };
    var points = at.Select((p,i) => string.Format(System.Globalization.CultureInfo.InvariantCulture,
        "\"{0}\":[{1:F2},{2:F2}]", fields[i], p.x, p.y));
    json.AppendFormat(System.Globalization.CultureInfo.InvariantCulture,
        "    {{\"edge\":\"{0}\",\"lengthMeters\":{1:F2},\"samples\":{2},\"widthMeters\":{3:F2},\"minRadiusM\":{4:F2},\"maxGradeDeg\":{5:F2},\"maxCrossDeg\":{6:F2},\"contacts\":{7},{8}}}{9}\n",
        id, length, vertices.Length, shoulder*2f, radius, grade, cross, blockers.Count, string.Join(",",points), main ? "" : ",");
    var mesh = new Mesh();
    mesh.vertices = vertices; mesh.normals = normals; mesh.colors32 = colors; mesh.triangles = faces;
    meshes.Add(mesh);
};

Func<Mesh,string> fingerprint = m =>
{
    using (var stream = new System.IO.MemoryStream())
    using (var writer = new System.IO.BinaryWriter(stream))
    using (var sha = System.Security.Cryptography.SHA256.Create())
    {
        foreach (var v in m.vertices) { writer.Write(v.x); writer.Write(v.y); writer.Write(v.z); }
        foreach (var n in m.normals) { writer.Write(n.x); writer.Write(n.y); writer.Write(n.z); }
        foreach (var c in m.colors32) { writer.Write(c.r); writer.Write(c.g); writer.Write(c.b); writer.Write(c.a); }
        foreach (var uv in m.uv) { writer.Write(uv.x); writer.Write(uv.y); }
        foreach (int t in m.triangles) writer.Write(t);
        writer.Flush();
        stream.Position = 0;
        return System.BitConverter.ToString(sha.ComputeHash(stream)).Replace("-", "").ToLowerInvariant();
    }
};

    return left.shader == right.shader && left.GetColor("_Tint") == right.GetColor("_Tint") &&
        left.GetFloat("_TileMeters") == right.GetFloat("_TileMeters") &&
        left.GetFloat("_VertexSurface") == right.GetFloat("_VertexSurface") &&

        left.GetTexture("_GravelNormal") == right.GetTexture("_GravelNormal") &&
        left.GetTexture("_Surface") == right.GetTexture("_Surface");
};

            go.AddComponent<MeshFilter>().sharedMesh = (Mesh)saved[2];
            go.AddComponent<MeshRenderer>().sharedMaterial = stoneMaterial;
            Undo.RegisterCreatedObjectUndo(go, "사암 노견");
        }
        Undo.FlushUndoRecordObjects();
        if (scene.isDirty && !UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene))
            throw new InvalidOperationException("씬을 저장하지 못했습니다.");
        Undo.CollapseUndoOperations(undoGroup);
    }
    catch
    {
        Undo.RevertAllDownToGroup(undoGroup);
        if (created) AssetDatabase.DeleteAsset(assetPath);
        else if (saved != null) foreach (var item in saved) AssetDatabase.SaveAssetIfDirty(item);
        throw;
    }
    return report;
}
finally
{
    foreach (var item in assets)
        if (item != null && !AssetDatabase.Contains(item)) UnityEngine.Object.DestroyImmediate(item);
}
