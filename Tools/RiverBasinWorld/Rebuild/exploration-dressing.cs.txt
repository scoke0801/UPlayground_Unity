var existing=GameObject.Find(rootName);
if (mode=="verify")
{
    if (existing==null || existing.transform.childCount!=groups.Length) throw new InvalidOperationException("탐험 배치 누락");
    var audit=new System.Text.StringBuilder();
    var drops=existing.GetComponentsInChildren<UPlayGround.DropItemActor>();
    if (drops.Length!=prizes.Length) throw new InvalidOperationException("보상 배치 수 불일치");
    var ids=new System.Collections.Generic.HashSet<string>();
    foreach(var drop in drops)
    {
        var id=drop.GetComponent<UPlayGround.Components.SceneEntityId>();
        if (drop.ItemData==null || drop.InteractionData==null || id==null || !id.HasGuid || !ids.Add(id.Guid)
            || drop.GetComponentInChildren<Collider>()==null) throw new InvalidOperationException("보상 데이터나 충돌 확인 "+drop.name);
        var p=drop.transform.position;
        audit.AppendLine(drop.name+" "+p.ToString("F1")+" "+drop.ItemData.name+" "+id.Guid);
    }
    foreach (Transform group in existing.transform)
    {
        int missing=0;
        foreach(var child in group.GetComponentsInChildren<Transform>(true))
            missing+=GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(child.gameObject);
        if(missing!=0) throw new InvalidOperationException("누락된 스크립트가 있습니다 "+group.name+" "+missing);
        audit.AppendLine(group.name+" / 배치 "+group.childCount+" / LOD "+group.GetComponentsInChildren<LODGroup>().Length
            +" / 콜라이더 "+group.GetComponentsInChildren<Collider>().Length+" / 누락 "+missing);
    }
    System.IO.File.WriteAllText(evidence+"dressing-verify.txt",audit.ToString());
    return audit.ToString();
}
if (existing!=null) throw new InvalidOperationException("탐험 배치가 이미 있습니다.");
var prefabs=new System.Collections.Generic.Dictionary<string,GameObject>();
foreach(var row in System.Linq.Enumerable.Concat(placements,prizes))
{
    string name=(string)row[1];
    if(prefabs.ContainsKey(name)) continue;
    string path=null;
    foreach(string guid in AssetDatabase.FindAssets(name+" t:Prefab",sourcePaths))
    {
        string candidate=AssetDatabase.GUIDToAssetPath(guid);
        if(System.IO.Path.GetFileNameWithoutExtension(candidate)!=name) continue;
        if(path!=null) throw new InvalidOperationException("프리팹 이름 중복 "+name);
        path=candidate;
    }
    var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(path);
    if(prefab==null || prefab.GetComponentInChildren<Renderer>(true)==null)
        throw new InvalidOperationException("프리팹 또는 렌더러 누락 "+name);
    prefabs.Add(name,prefab);
}
var interaction=AssetDatabase.LoadAssetAtPath<UPlayGround.Data.Actor.InteractableActorSO>("Assets/10.Datas/Actor/Interaction/DropItem.asset");

foreach(var row in prizes)
    if(AssetDatabase.LoadAssetAtPath<UPlayGround.Data.Item.ItemSO>((string)row[4])==null)
        throw new InvalidOperationException("아이템 누락 "+row[4]);
if(mode=="dry") return "프리팹 "+prefabs.Count+"종 / 환경 "+placements.Length+"개 / 획득 "+prizes.Length+"개";
if(mode!="apply") throw new InvalidOperationException("적용 모드를 확인하세요.");
var currentLake=GameObject.Find("남부 호수 물가와 섬");
var shifts=System.IO.File.ReadAllLines(backup+"lake-prop-shifts.csv");
Undo.IncrementCurrentGroup();
int undo=Undo.GetCurrentGroup();
Undo.SetCurrentGroupName("호수·동부·설산 탐험 배치");
try
{
    foreach(string line in shifts)
    {
        var parts=line.Split(',');
        var t=currentLake.transform.GetChild(int.Parse(parts[0])).GetChild(int.Parse(parts[1]));
        Undo.RecordObject(t,"섬 식생과 암반 지면 정렬");
        t.position+=Vector3.up*float.Parse(parts[2],System.Globalization.CultureInfo.InvariantCulture);
        PrefabUtility.RecordPrefabInstancePropertyModifications(t);
    }
    var root=new GameObject(rootName);
    Undo.RegisterCreatedObjectUndo(root,"탐험 배치");
    var parents=new Transform[groups.Length];
    for(int i=0;i<groups.Length;i++)
    {
        var group=new GameObject(groups[i]);
        group.transform.SetParent(root.transform,false);
        parents[i]=group.transform;
    }
    Func<GameObject,Bounds> bounds = obj =>
    {
        var renderers=obj.GetComponentsInChildren<Renderer>();
        if(renderers.Length==0) throw new InvalidOperationException("활성 렌더러 누락 "+obj.name);
        var b=renderers[0].bounds;
        foreach(var renderer in renderers)
        {
            if(Array.Exists(renderer.sharedMaterials,m=>m==null || m.shader==null))
                throw new InvalidOperationException("재질 누락 "+obj.name);
            b.Encapsulate(renderer.bounds);
        }
        return b;
    };
    foreach(var row in placements)
    {
        var obj=(GameObject)PrefabUtility.InstantiatePrefab(prefabs[(string)row[1]],parents[(int)row[0]]);
        Undo.RegisterCreatedObjectUndo(obj,"탐험 지형 표지물");
        float x=(float)row[2], z=(float)row[3], width=(float)row[4];
        obj.transform.SetPositionAndRotation(new Vector3(x,height(x,z),z),Quaternion.Euler(0,(float)row[5],0));
        Bounds b=bounds(obj);
        if(width>0)
        {
            obj.transform.localScale*=width/Mathf.Max(b.size.x,b.size.z);
            b=bounds(obj);
        }
        float low=height(x,z);
        foreach(float bx in new[]{b.min.x,b.max.x})
        foreach(float bz in new[]{b.min.z,b.max.z})
            low=Mathf.Min(low,height(bx,bz));
        obj.transform.position+=Vector3.up*(Mathf.Lerp(height(x,z),low,0.5f)-b.min.y-b.size.y*(float)row[6]);
    }
    foreach(var row in prizes)
    {
        float x=(float)row[2], z=(float)row[3];
        var item=AssetDatabase.LoadAssetAtPath<UPlayGround.Data.Item.ItemSO>((string)row[4]);
        var marker=new GameObject("발견 · "+item.name);
        Undo.RegisterCreatedObjectUndo(marker,"탐험 발견 보상");
        marker.transform.SetParent(parents[(int)row[0]],false);
        marker.transform.position=new Vector3(x,height(x,z)+0.14f,z);
        var visual=(GameObject)PrefabUtility.InstantiatePrefab(prefabs[(string)row[1]],marker.transform);
        Undo.RegisterCreatedObjectUndo(visual,"보상 모델");
        visual.transform.localPosition=Vector3.zero;
        var b=bounds(visual);
        visual.transform.position+=Vector3.up*(marker.transform.position.y-b.min.y);
        var collider=Undo.AddComponent<SphereCollider>(marker);
        collider.radius=0.43f;
        collider.center=Vector3.up*0.28f;
        collider.isTrigger=true;
        var id=Undo.AddComponent<UPlayGround.Components.SceneEntityId>(marker);
        id.EditorSetGuid((string)row[5]);
        Undo.AddComponent<UPlayGround.DropItemActor>(marker).Init(item,1,interaction);
    }
    // 수변의 자갈턱과 그 위 식생을 구분하고 채석 사면에는 회색 암반을 드러낸다.
    foreach(var tile in new[]{grid[1,1],grid[1,2],grid[1,5],grid[2,5],grid[4,6]})
    {
        var data=tile.terrainData;
        var pos=tile.transform.position;
        var layers=data.terrainLayers;
        int meadow=Array.FindIndex(layers,l=>l!=null && l.name=="Meadow");
        int shore=Array.FindIndex(layers,l=>l!=null && l.name=="Shore");
        int rock=Array.FindIndex(layers,l=>l!=null && l.name=="Rock");
        if(meadow<0 || shore<0 || rock<0) throw new InvalidOperationException("기존 지면 레이어 누락 "+tile.name);
        var alpha=data.GetAlphamaps(0,0,data.alphamapWidth,data.alphamapHeight);
        bool painted=false;
        for(int az=0;az<data.alphamapHeight;az++)
        for(int ax=0;ax<data.alphamapWidth;ax++)
        {
            float x=pos.x+(ax+0.5f)/data.alphamapWidth*512;
            float z=pos.z+(az+0.5f)/data.alphamapHeight*512;
            float weight;
            int target;
            float noise=Mathf.PerlinNoise(x*0.055f,z*0.055f);
            if(pos.z>=2048)
            {
                float ex=(x-3200)/63f, ez=(z-2213)/74f;
                float radial=Mathf.Sqrt(ex*ex+ez*ez);
                weight=Mathf.SmoothStep(0,0.94f,Mathf.Clamp01((1.15f-radial+(noise-0.5f)*0.15f)/0.48f));
                target=rock;
            }
            else
            {
                bool west=x<2000;
                float cx=west?1104:2829, cz=west?834:998;
                float ex=(x-cx)/153f, ez=(z-cz)/105f;
                float radial=Mathf.Sqrt(ex*ex+ez*ez);
                float above=height(x,z)-(west?15:24);
                if(above<0.25f || radial>1.18f) continue;
                weight=Mathf.SmoothStep(0,0.73f,Mathf.Clamp01((1.18f-radial)/0.31f));
                target=above<2.0f+(noise-0.5f)*1.7f?shore:meadow;
                float dx=0.5f*(height(x+1,z)-height(x-1,z));
                float dz=0.5f*(height(x,z+1)-height(x,z-1));
                if(dx*dx+dz*dz>0.36f && above>2.5f) target=rock;
            }
            if(weight<0.01f) continue;
            for(int layer=0;layer<layers.Length;layer++) alpha[az,ax,layer]*=1-weight;
            alpha[az,ax,target]+=weight;
            painted=true;
        }
        if(!painted) continue;
        Undo.RegisterCompleteObjectUndo(data,"섬 지면과 채석 단구 재질");
        data.SetAlphamaps(0,0,alpha);
        EditorUtility.SetDirty(data);
    }
    UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
    if(!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene)) throw new InvalidOperationException("씬 저장 실패");
    foreach(var tile in new[]{grid[1,1],grid[1,2],grid[1,5],grid[2,5],grid[4,6]})
        AssetDatabase.SaveAssetIfDirty(tile.terrainData);
    Undo.CollapseUndoOperations(undo);
}
catch
{
    Undo.RevertAllDownToGroup(undo);
    throw;
}
return "환경 "+placements.Length+"개 / 배치 보상 "+prizes.Length+"개 / 기존 섬 장식 재정렬 "+shifts.Length+"개";
