// 위치·상대 높이·배후 폭을 같은 곡선에서 읽어 메시와 Terrain의 경계를 맞춘다.
Func<Vector4[],Vector4[]> sample=points=>{
    var result=new System.Collections.Generic.List<Vector4>();
    for(int i=0;i<points.Length-1;i++){
        var a=points[Mathf.Max(0,i-1)];var b=points[i];var c=points[i+1];var d=points[Mathf.Min(points.Length-1,i+2)];
        int count=Mathf.CeilToInt(Vector2.Distance(new Vector2(b.x,b.y),new Vector2(c.x,c.y))/2.75f);
        for(int j=0;j<count;j++){
            float f=j/(float)count;
            result.Add(.5f*(2*b+(-a+c)*f+(2*a-5*b+4*c-d)*f*f+(-a+3*b-3*c+d)*f*f*f));
        }
    }
    result.Add(points.Last());return result.ToArray();
};
var ridges=new[]{sample(east),sample(west)};
var planPoints=ridges.Select(items=>items.Select(p=>new Vector2(p.x,p.y)).ToArray()).ToArray();
var normal=ridges.Select((ridge,id)=>ridge.Select((p,i)=>{
    var line=planPoints[id];
    var d=(line[Mathf.Min(i+1,line.Length-1)]-line[Mathf.Max(0,i-1)]).normalized;
    return new Vector2(-d.y,d.x);
}).ToArray()).ToArray();
var road=sceneRoots.SelectMany(r=>r.GetComponentsInChildren<MeshFilter>()).Where(f=>f.name.StartsWith("WB_P04_Road_R4")||f.name=="WB_P04_Road_R5")
    .SelectMany(f=>f.sharedMesh.vertices.Select(v=>f.transform.TransformPoint(v)))
    .Where(v=>v.x>3580&&v.z>1010&&v.z<1545)
    .Select(v=>new Vector2(v.x,v.z)).Where((p,i)=>i%3==0).ToArray();
Func<Vector2,int,float> formation=(p,r)=>{
    float distance=float.PositiveInfinity,side=0;
    var chain=planPoints[r];
    var row=ridges[r][0];
    for(int i=1;i<chain.Length;i++){
        var segment=chain[i]-chain[i-1];
        float t=Mathf.Clamp01(Vector2.Dot(p-chain[i-1],segment)/segment.sqrMagnitude);
        var offset=p-Vector2.Lerp(chain[i-1],chain[i],t);
        float score=offset.sqrMagnitude;
        if(score>=distance)continue;
        distance=score;
        row=Vector4.Lerp(ridges[r][i-1],ridges[r][i],t);
        side=Vector2.Dot(offset,new Vector2(-segment.y,segment.x).normalized);
    }
    float n=Mathf.PerlinNoise(p.x*.032f,p.y*.037f);
    float across = side>=0 ? .095f*(1-gate(side,22+13*n)) :
        Mathf.Lerp(.095f,1,gate(-side,16+r*2f))*(1-gate(-side-row.w*.44f,row.w*.79f));
    return row.z*across;
};
var trails=new[]{
    new Vector2(3701.5f,1523),new Vector2(3707,1496),new Vector2(3720,1451),new Vector2(3710,1409),new Vector2(3716,1373),
    new Vector2(3727,1333),new Vector2(3724,1290),new Vector2(3734,1258),new Vector2(3752,1227),new Vector2(3761,1196),
    new Vector2(3785,1169),new Vector2(3812,1151),new Vector2(3830,1141),new Vector2(3830,1124)

var duneTrail=new[]{new Vector2(3716,1373),new Vector2(3742,1350),new Vector2(3770,1323),new Vector2(3763,1285),
    new Vector2(3779,1234),new Vector2(3801,1207),new Vector2(3806,1176),new Vector2(3812,1151)};
Func<Vector2,Vector2[],float> trailDistance=(p,waypoints)=>{
    float nearest=float.PositiveInfinity;
    for(int i=1;i<waypoints.Length;i++){
        var line=waypoints[i]-waypoints[i-1];
        var foot=waypoints[i-1]+line*Mathf.Clamp01(Vector2.Dot(p-waypoints[i-1],line)/line.sqrMagnitude);
        nearest=Mathf.Min(nearest,(p-foot).sqrMagnitude);
    }
    return Mathf.Sqrt(nearest);
};
Func<Vector2,float> nearSite=p=>{
    float x=p.x-3828, z=p.y-1128;
    return 1-gate(new Vector2(x/37,z/31).magnitude-1,.36f);
};
var oldProps=section.Find("모랫길 · 드러난 사암층과 돌 표식");
var existingHeights=oldProps.Cast<Transform>().ToDictionary(p=>p,p=>h(p.position.x,p.position.z));

var deltaHeights=new float[res,res];

int sandstone=Array.FindIndex(land.terrainLayers,l=>l.name=="Sandstone");
if(sand<0||sandstone<0)throw new InvalidOperationException("모래 레이어가 누락됐습니다.");
int samples=0;
for(int z=1;z<res-1;z++)for(int x=1;x<res-1;x++){
    var p=new Vector2(origin.x+x*512f/(res-1),origin.z+z*512f/(res-1));
    if(p.y>1511)continue;
    float border=gate(x,34)*gate(z,55)*gate(res-1-x,34)*gate(1512-p.y,31);
    float site=nearSite(p);
    float uplift=Mathf.Max(formation(p,0),formation(p,1));
    if(road.Length>0&&uplift>0&&p.x<3748){
        float dist=Mathf.Sqrt(road.Min(q=>(q-p).sqrMagnitude));
        uplift*=gate(dist-14,25);
    }
    deltaHeights[z,x]=uplift*border;
    var ground=Mathf.Lerp(restored[z,x],previous[z,x],site)+deltaHeights[z,x]/land.size.y;
    if(Mathf.Abs(ground-restored[z,x])>.012f/land.size.y)samples++;
    restored[z,x]=ground;
}
for(int z=1;z<alphaRes-1;z++)for(int x=1;x<alphaRes-1;x++){
    var p=new Vector2(origin.x+x*512f/(alphaRes-1),origin.z+z*512f/(alphaRes-1));
    if(p.y>1511)continue;
    var move=deltaHeights[Mathf.RoundToInt(z*(res-1f)/(alphaRes-1)),Mathf.RoundToInt(x*(res-1f)/(alphaRes-1))];
    int layers=alpha.GetLength(2);
    float preserve=nearSite(p);
    float noise=Mathf.PerlinNoise(p.x*.058f,p.y*.052f);
    float trail=trailDistance(p,trails),alt=trailDistance(p,duneTrail);
    for(int k=0;k<layers;k++)alpha[z,x,k]=Mathf.Lerp(alpha[z,x,k],currentAlpha[z,x,k],preserve);
    float wash=(1-gate(trail,12+5*noise))*(.16f+.25f*noise);
    float branch=(1-gate(alt,8+4*noise))*(.12f+.19f*noise);
    float edge=gate(x,8)*gate(z,8)*gate(alphaRes-1-x,8)*gate(1512-p.y,31);
    float layer=alpha[z,x,sand]*Mathf.Max(Mathf.Max(wash,branch),gate(move-1.6f,26)*(.38f+noise*.48f))*edge;
    alpha[z,x,sand]-=layer;
    alpha[z,x,sandstone]+=layer;
}

var bevel=new[]{13.3f,5.5f,1.7f,.6f,-1.8f,-.3f,-2.1f,-1.15f,-3.06f,-1.5f,-4.4f,-6.3f,-9.5f,-19f};

        var rows=new Vector3[chain.Length,levels.Length];

        for(int i=0;i<chain.Length;i++){
            float run=total[i];

                var pos=at+lateral*xz;

                if(j==0||j==levels.Length-1)y=baseH(pos.x,pos.y)+(j==0?-1.4f:vertical*.92f);

        var worldCenter=dressing.position+marks[i].center;
