Func<float,float,float,float> smooth = (a,b,v) => Mathf.SmoothStep(0,1,Mathf.InverseLerp(a,b,v));
Func<Vector2,float> lift = p => {
    float z = p.y;
    float envelope = smooth(318,416,p.x) * smooth(1940,2010,z) * (1-smooth(2398,2520,z));
    float slope = 1-smooth(835,1005,p.x);
    float face = 1-smooth(848+5*Mathf.Sin(z*.067f),878+6*Mathf.Sin(z*.056f),p.x);
    float waterfall = 1-smooth(29,91,Mathf.Abs(z-2188));
    return lakeRise*envelope*Mathf.Lerp(slope,face,waterfall);
};

    var p0=profile[Mathf.Max(0,segment-1)];var p1=profile[segment];

    return b;
};

Func<Vector2,Vector4> channel = p => {
    var result=Vector4.zero;float best=10000;
    if(p.x<766||p.x>1603||p.y<2085||p.y>2260)return new Vector4(100,0,0,0);
    for(int i=1;i<rowArray.Length;i++){
        var a=rowArray[i-1];var b=rowArray[i];
        if(Mathf.Abs(p.x-a.x)>42||Mathf.Abs(p.y-a.y)>42)continue;
        var dir=xy(b-a);float f=Mathf.Clamp01(Vector2.Dot(p-xy(a),dir)/dir.sqrMagnitude);
        var c=Vector4.Lerp(a,b,f);float d=(p-xy(c)).sqrMagnitude;
        if(d<best){best=d;result=c;}
    }
    return new Vector4(Mathf.Sqrt(best),result.z,result.w,result.x);
};
