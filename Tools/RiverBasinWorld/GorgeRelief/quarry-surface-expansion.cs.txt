Func<Vector3[], Vector2, Vector2> nearest = (line, p) =>
{
    var result = new Vector2(float.PositiveInfinity, 0);
    for (int i=1; i<line.Length; i++)
    {
        var first = new Vector2(line[i-1].x, line[i-1].z);
        var delta = new Vector2(line[i].x, line[i].z) - first;
        float t = Mathf.Clamp01(Vector2.Dot(p-first, delta)/delta.sqrMagnitude);
        float distance = Vector2.Distance(p, first+delta*t);
        if (distance < result.x) result = new Vector2(distance, Mathf.Lerp(line[i-1].y,line[i].y,t));
    }
    return result;
};

Action<string,Transform,Material,System.Collections.Generic.List<Vector3>,System.Collections.Generic.List<int>,bool> attach =
    (name,parent,material,vertices,indices,collidable) =>
{
    if (indices.Count==0) return;
    var mesh=new Mesh {name=name};
    mesh.SetVertices(vertices);
    mesh.SetTriangles(indices,0);
    var uv=new Vector2[vertices.Count];
    for (int i=0;i<uv.Length;i+=4)
    {
        var normal=Vector3.Cross(vertices[i+1]-vertices[i],vertices[i+2]-vertices[i]);
        float nx=Mathf.Abs(normal.x), ny=Mathf.Abs(normal.y), nz=Mathf.Abs(normal.z);
        for(int j=i;j<i+4;j++)
        {
            var v=vertices[j];
            uv[j]=(ny>=nx&&ny>=nz?new Vector2(v.x,v.z):nx>nz?new Vector2(v.z,v.y):new Vector2(v.x,v.y))*.35f;
        }
    }
    mesh.uv=uv;
    mesh.RecalculateNormals();
    mesh.RecalculateBounds();
    meshes.Add(mesh);
    var node=make(name,parent,anchor);
    node.AddComponent<MeshFilter>().sharedMesh=mesh;
    node.AddComponent<MeshRenderer>().sharedMaterial=material;
    if (collidable) node.AddComponent<MeshCollider>().sharedMesh=mesh;
};

    var surface=make("절개 단구와 인양 작업장",root.transform,anchor).transform;
    var blocksV=new System.Collections.Generic.List<Vector3>(); var blocksI=new System.Collections.Generic.List<int>();
    var random=new System.Random(909);
    for(int row=0;row<benches.Length;row++)
    {
        var line=benches[row];
        for(int col=0;col<12;col++)
        {
            float t=(col+.5f)/12;
            var center=Vector3.Lerp(line[0],line[2],t);
            center.z-=3.3f+(float)random.NextDouble()*1.6f;
            float halfWidth=1.9f+(float)random.NextDouble()*.38f;
            float depth=1.5f+(float)random.NextDouble();
            float high=1.35f+(float)random.NextDouble()*2.15f;
            var a=center+new Vector3(-halfWidth,-.45f,0); var f=center+new Vector3(halfWidth,-.45f,-.35f);
            var e=a+new Vector3(.33f,high,-.25f); var g=f+new Vector3(-.21f,high+.16f,-.13f);
            var dr=new Vector3(.25f,-.1f,-depth);
            quad(blocksV,blocksI,a,f,g,e,Vector3.forward);
            quad(blocksV,blocksI,e,g,g+dr,e+dr,Vector3.up);
            quad(blocksV,blocksI,f,f+dr,g+dr,g,Vector3.right);
            quad(blocksV,blocksI,a+dr,a,e,e+dr,Vector3.left);
            quad(blocksV,blocksI,f+dr,a+dr,e+dr,g+dr,Vector3.back);
        }
    }
    attach("Dng_Wall_QuarryExtractionFaces",surface,rock,blocksV,blocksI,true);
    var trackV=new System.Collections.Generic.List<Vector3>();
    var trackI=new System.Collections.Generic.List<int>();
    var trackMaterial=new Material(rock) {name="채석장 운반로 · 쇄석"};
    trackMaterial.SetColor("_BaseColor",new Color(.55f,.49f,.38f));
    for(int routeIndex=0;routeIndex<roads.Length;routeIndex++)
    {
        var line=roads[routeIndex];
        for(int i=1;i<line.Length;i++)
        {
            var start=new Vector2(line[i-1].x,line[i-1].z);
            var end=new Vector2(line[i].x,line[i].z);
            var dir=(end-start).normalized;
            var right=new Vector2(-dir.y,dir.x);
            int steps=Mathf.CeilToInt(Vector2.Distance(start,end)*2);
            for(int k=0;k<steps;k++)
            {
                var a=Vector2.Lerp(start,end,(float)k/steps);
                var f=Vector2.Lerp(start,end,(float)(k+1)/steps);
                if(routeIndex>=2 && i==line.Length-1 && routeIndex!=5 && Vector2.Distance(f,end)<2) continue;
                float halfA=1.07f+Mathf.PerlinNoise(a.x*.55f,a.y*.55f)*.84f;
                float halfB=1.07f+Mathf.PerlinNoise(f.x*.55f,f.y*.55f)*.84f;
                for(int strip=-2;strip<2;strip++)
                {
                    var p=a+right*halfA*strip*.5f; var q=a+right*halfA*(strip+1)*.5f;
                    var u=f+right*halfB*strip*.5f; var v=f+right*halfB*(strip+1)*.5f;
                    quad(trackV,trackI,new Vector3(p.x,ground(p)+.045f,p.y),new Vector3(q.x,ground(q)+.045f,q.y),
                        new Vector3(v.x,ground(v)+.045f,v.y),new Vector3(u.x,ground(u)+.045f,u.y),Vector3.up);
                }
            }
        }
    }
    attach("쇄석 운반길",surface,trackMaterial,trackV,trackI,false);
    Func<float,float,Vector3> onGround=(x,z)=>new Vector3(x,ground(new Vector2(x,z)),z);
    var hoistBase=onGround(3232,2261.5f);
    var hoist=make("목재 인양틀",surface,hoistBase).transform;
    var timberMaterial=timber.GetComponentInChildren<MeshRenderer>().sharedMaterial;
    Action<string,Vector3,Vector3,float,Material> beam = (name,p,q,thickness,material) =>
    {
        var o=GameObject.CreatePrimitive(PrimitiveType.Cube);
        o.name=name; o.transform.SetParent(hoist);
        o.transform.position=(p+q)*.5f;
        o.transform.rotation=Quaternion.FromToRotation(Vector3.up,(q-p).normalized);
        o.transform.localScale=new Vector3(thickness,Vector3.Distance(p,q),thickness);
        o.GetComponent<MeshRenderer>().sharedMaterial=material;
    };
    var mast=hoistBase+Vector3.up*8;
    var tip=mast+new Vector3(-4,2,-1);
    beam("인양 기둥",hoistBase-Vector3.up*.5f,mast,.65f,timberMaterial);
    beam("인양 붐",mast+new Vector3(2,-1,.5f),tip,.49f,timberMaterial);
    foreach(var offset in new[]{new Vector3(2.7f,0,2.6f),new Vector3(2.7f,0,-2.7f),new Vector3(-2.2f,0,1.8f)})
        beam("기둥 버팀",onGround(hoistBase.x+offset.x,hoistBase.z+offset.z)-Vector3.up*.22f,mast-Vector3.up*2,.36f,timberMaterial);
    beam("인양 줄",tip,tip-Vector3.up*5,.065f,timberMaterial);
    beam("매달린 판석",tip-Vector3.up*5,tip-Vector3.up*5.48f,1.25f,rock);
    prop(cart,surface,onGround(3221,2257),Quaternion.Euler(0,82,0));
    prop(cart,surface,onGround(3269.8f,2207),Quaternion.Euler(0,176,0));

    Physics.SyncTransforms();
    AssetDatabase.CreateAsset(work,terrainPath);
    AssetDatabase.CreateAsset(meshes[0],meshPath);
    for(int i=1;i<meshes.Count;i++) AssetDatabase.AddObjectToAsset(meshes[i],meshPath);
    AssetDatabase.AddObjectToAsset(trackMaterial,meshPath);
    AssetDatabase.SaveAssetIfDirty(work);
    AssetDatabase.SaveAssetIfDirty(meshes[0]);
    if(!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene))
        throw new InvalidOperationException("프리뷰 씬 저장 실패");
    Undo.CollapseUndoOperations(undo);
    return "지상 채석면 3단, 연결 갱구 2곳, 독립 동굴 1곳 저장";
}
catch
{
    Undo.RevertAllDownToGroup(undo);
    throw;
