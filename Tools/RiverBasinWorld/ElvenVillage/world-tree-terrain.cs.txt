const string mode = "apply";
if (scene.isDirty || GameObject.Find(label)) throw new Exception("기존 씬 변경 또는 산정호 작업이 있습니다.");
var terrainFolder=world+"/Production/Terrain";
var originals=new Dictionary<Terrain,TerrainData>();
var clones=new Dictionary<Terrain,TerrainData>();
var created=new List<string>();
var report=new List<string>();
var copies=new[] { new Vector2Int(0,3),new Vector2Int(1,3),new Vector2Int(0,4),new Vector2Int(1,4),
    new Vector2Int(2,4),new Vector2Int(3,4) };
foreach(var id in copies){
    var t=tiles[id.x,id.y];var data=t.terrainData;var origin=t.GetPosition();
    var dst=terrainFolder+"/"+t.name+"_WorldTreeBasin.asset";
    if(AssetDatabase.LoadMainAssetAtPath(dst))throw new Exception("이미 존재하는 파생 지형 "+dst);
    var res=data.heightmapResolution;float step=data.size.x/(res-1);
    var heights=data.GetHeights(0,0,res,res);
    float maxRise=0,maxCut=0;int changed=0;
    for(int iz=0;iz<res;iz++)for(int ix=0;ix<res;ix++){
        var p=new Vector2(origin.x+ix*step,origin.z+iz*step);
        var old=origin.y+heights[iz,ix]*data.size.y;
        var y=shape(p,old);var d=y-old;
        if(Mathf.Abs(d)<.0005f)continue;
        changed++;maxRise=Mathf.Max(maxRise,d);maxCut=Mathf.Max(maxCut,-d);
        heights[iz,ix]=(y-origin.y)/data.size.y;
    }
    if(changed==0)continue;
    report.Add(t.name+" 높이 "+changed+"개, 성토 "+maxRise.ToString("F1")+"m, 절토 "+maxCut.ToString("F1")+"m");
    if(mode!="apply")continue;
    var clone=UnityEngine.Object.Instantiate(data);clone.name=t.name+"_WorldTreeBasin";
    clone.SetHeights(0,0,heights);
    originals.Add(t,data);clones.Add(t,clone);
}
if(mode!="apply")return string.Join("\n",report);
var reanchored=new Dictionary<Transform,Vector3>();
var originalRoads=new Dictionary<MeshFilter,Mesh>();
var oldTerrain=(Func<float,float,float>)((x,z)=>{
    var t=tiles[Mathf.Clamp(Mathf.FloorToInt(x/512),0,7),Mathf.Clamp(Mathf.FloorToInt(z/512),0,7)];
    TerrainData data;
    if(!originals.TryGetValue(t,out data))data=t.terrainData;
    var o=t.GetPosition();
    return o.y+data.GetInterpolatedHeight((x-o.x)/data.size.x,(z-o.z)/data.size.z);
});
try{
    if(!AssetDatabase.IsValidFolder(product))AssetDatabase.CreateFolder(world+"/Production","WorldTreeBasin");
    // 수평 위치와 원래의 접지 오프셋은 유지하고, 물길과 광장만 기존 수목을 걷어낸다.
    foreach(var pair in clones){
        var t=pair.Key;var data=originals[t];var clone=pair.Value;var o=t.GetPosition();
        var trees=new List<TreeInstance>();int removed=0, moved=0;
        Func<Vector2,bool> clear = p => {
            var d=channel(p);
            bool stream=d.z>0 && d.x<d.z+2.2f && p.x>822 && p.x<1563;
            bool court=(p-village).sqrMagnitude<20f*20f;
            bool cliff=p.x>845&&p.x<899&&Mathf.Abs(p.y-2188)<31;
            return stream||court||cliff;
        };
        foreach(var instance in data.treeInstances){
            var tr=instance;var p=o+Vector3.Scale(data.size,tr.position);
            if(clear(new Vector2(p.x,p.z))){removed++;continue;}
            float before=data.GetInterpolatedHeight(tr.position.x,tr.position.z);
            float after=clone.GetInterpolatedHeight(tr.position.x,tr.position.z);
            if(Mathf.Abs(after-before)>.015f){
                tr.position=new Vector3(tr.position.x,Mathf.Clamp01(tr.position.y+(after-before)/data.size.y),tr.position.z);moved++;
            }
            trees.Add(tr);
        }
        clone.SetTreeInstances(trees.ToArray(),false);
        var res=data.alphamapResolution;var layers=clone.terrainLayers;
        int rock=Array.FindIndex(layers,l=>l.name=="Rock"), shore=Array.FindIndex(layers,l=>l.name=="Shore");
        var alpha=clone.GetAlphamaps(0,0,res,res);
        for(int z=0;z<res;z++)for(int x=0;x<res;x++){
            var p=new Vector2(o.x+(x+.5f)*data.size.x/res,o.z+(z+.5f)*data.size.z/res);
            var r=channel(p);
            float noise=Mathf.PerlinNoise(p.x*.083f,p.y*.082f);
            float face=(1-smooth(20,88,Mathf.Abs(p.y-2188)))*smooth(835,850,p.x)*(1-smooth(879,907,p.x));
            float channelWeight=r.z>0 ? (1-smooth(r.z-1,r.z+7,r.x))*smooth(816,842,p.x)*(1-smooth(1550,1570,p.x)):0;
            float cliff=face*Mathf.Lerp(.54f,.91f,noise);
            float gravel=channelWeight*Mathf.Lerp(.53f,.84f,noise)*(1-cliff);
            if(cliff+gravel<=.005f||rock<0||shore<0)continue;
            for(int layer=0;layer<layers.Length;layer++)alpha[z,x,layer]*=1-cliff-gravel;
            alpha[z,x,rock]+=cliff;alpha[z,x,shore]+=gravel;
        }
        clone.SetAlphamaps(0,0,alpha);
        int dr=clone.detailResolution;
        int x0=Mathf.Clamp(Mathf.FloorToInt((795-o.x)*dr/data.size.x),0,dr);
        int x1=Mathf.Clamp(Mathf.CeilToInt((1575-o.x)*dr/data.size.x),0,dr);
        int z0=Mathf.Clamp(Mathf.FloorToInt((2090-o.z)*dr/data.size.z),0,dr);
        int z1=Mathf.Clamp(Mathf.CeilToInt((2340-o.z)*dr/data.size.z),0,dr);
        if(x1>x0&&z1>z0&&clone.detailPrototypes.Length>0){
            var mask=new bool[z1-z0,x1-x0];
            for(int z=z0;z<z1;z++)for(int x=x0;x<x1;x++)
                mask[z-z0,x-x0]=clear(new Vector2(o.x+(x+.5f)*data.size.x/dr,o.z+(z+.5f)*data.size.z/dr));
            for(int i=0;i<clone.detailPrototypes.Length;i++){
                var detail=clone.GetDetailLayer(x0,z0,x1-x0,z1-z0,i);
                for(int z=0;z<z1-z0;z++)for(int x=0;x<x1-x0;x++)if(mask[z,x])detail[z,x]=0;
                clone.SetDetailLayer(x0,z0,i,detail);
            }
        }
        var dest=terrainFolder+"/"+clone.name+".asset";
        AssetDatabase.CreateAsset(clone,dest);created.Add(dest);
        t.terrainData=clone;t.GetComponent<TerrainCollider>().terrainData=clone;t.Flush();
        AssetDatabase.SaveAssetIfDirty(clone);
        report.Add(t.name+" 수목 정리 "+removed+", 재접지 "+moved);
    }
    // 기존 호수 주변 장식과 R1b 흙길은 현재 표고 변화량만큼 이동한다.
    foreach(var root in scene.GetRootGameObjects())foreach(var tr in root.GetComponentsInChildren<Transform>(true)){
        if(tr.name=="숲 연못의 섬" || (tr.parent && tr.parent.name=="숲 연못 동쪽 식생")){
            var p=tr.position;reanchored.Add(tr,p);p.y+=ground(p.x,p.z)-oldTerrain(p.x,p.z);tr.position=p;
        }
        if(tr.name!="WB_P04_Road_R1b"&&tr.name!="WB_P04_Road_R1")continue;
        var f=tr.GetComponent<MeshFilter>();var src=f.sharedMesh;
        var mesh=UnityEngine.Object.Instantiate(src);mesh.name=tr.name+"_WorldTreeBasin";
        var verts=mesh.vertices;var xz=new Vector3[verts.Length];
        for(int i=0;i<verts.Length;i++){
            var p=tr.TransformPoint(verts[i]);xz[i]=p;
            if(tr.name=="WB_P04_Road_R1"&&Vector2.Distance(new Vector2(p.x,p.z),new Vector2(1275,2199))<15)continue;
            p.y+=ground(p.x,p.z)-oldTerrain(p.x,p.z);
            verts[i]=tr.InverseTransformPoint(p);
        }
        mesh.vertices=verts;
        if(tr.name=="WB_P04_Road_R1"){
            var triangles=mesh.triangles;
            var keep=new List<int>();
            for(int i=0;i<triangles.Length;i+=3){
                var p=(xz[triangles[i]]+xz[triangles[i+1]]+xz[triangles[i+2]])/3;
                var r=channel(new Vector2(p.x,p.z));
                if(Vector2.Distance(new Vector2(p.x,p.z),new Vector2(1275,2199))<14 && r.x<r.z+.45f)continue;
                keep.Add(triangles[i]);keep.Add(triangles[i+1]);keep.Add(triangles[i+2]);
            }
            mesh.triangles=keep.ToArray();
        }
        mesh.RecalculateNormals();mesh.RecalculateBounds();
        var dest=product+"/"+mesh.name+".asset";
        AssetDatabase.CreateAsset(mesh,dest);created.Add(dest);
        originalRoads.Add(f,src);f.sharedMesh=mesh;
        AssetDatabase.SaveAssetIfDirty(mesh);
    }
    UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
    return string.Join("\n",report);
}catch{
    foreach(var pair in originals){pair.Key.terrainData=pair.Value;pair.Key.GetComponent<TerrainCollider>().terrainData=pair.Value;pair.Key.Flush();}
    foreach(var pair in reanchored)pair.Key.position=pair.Value;
    foreach(var pair in originalRoads)pair.Key.sharedMesh=pair.Value;
    foreach(var path in created)AssetDatabase.DeleteAsset(path);
    throw;
}
