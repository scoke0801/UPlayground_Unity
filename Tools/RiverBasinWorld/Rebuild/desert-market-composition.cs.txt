Action<System.Collections.Generic.List<Vector3>,System.Collections.Generic.List<int>,Vector3,Vector3,Vector3,Vector3,Vector3> quad=(verts,tris,a,b,c2,d,normal)=>{
    int i=verts.Count;verts.AddRange(new[]{a,b,c2,d});
    if(Vector3.Dot(Vector3.Cross(b-a,c2-a),normal)>0)tris.AddRange(new[]{i,i+1,i+2,i,i+2,i+3});
    else tris.AddRange(new[]{i,i+2,i+1,i,i+3,i+2});
};
Func<Transform,string,System.Collections.Generic.List<Vector3>,System.Collections.Generic.List<int>[],Material[],bool,GameObject> makeMesh=(parent,name,vertices,indices,mats,collides)=>{
    var go=child(parent,name);var origin=vertices[0];go.transform.position=origin;
    var m=new Mesh{name=name};
    m.vertices=vertices.Select(v=>v-origin).ToArray();
    m.uv=vertices.Select(v=>new Vector2((v.x+v.z*.7f)*.33f,(v.z+v.y*1.1f)*.38f)).ToArray();
    m.subMeshCount=indices.Length;
    for(int i=0;i<indices.Length;i++)m.SetTriangles(indices[i],i);
    m.RecalculateNormals();m.RecalculateTangents();m.RecalculateBounds();
    go.AddComponent<MeshFilter>().sharedMesh=m;
    go.AddComponent<MeshRenderer>().sharedMaterials=mats;
    if(collides){go.layer=groundLayer;go.AddComponent<MeshCollider>().sharedMesh=m;}
    assets.Add(m);return go;
};
Func<Transform,GameObject,Vector3,float,Vector3,GameObject> place=(parent,source,where,yaw,size)=>{

    go.transform.SetPositionAndRotation(where,Quaternion.Euler(0,yaw,0)*source.transform.localRotation);
    go.transform.localScale=Vector3.Scale(go.transform.localScale,size);
    return go;
};

    }
