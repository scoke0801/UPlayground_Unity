Action<Vector3,Vector3,Vector3,Vector3,Vector3,Vector3> quad=(origin,a,b,c,d,normal)=>

            new Vector2(Mathf.Abs(normal.x)>.65f?pos.z:pos.x,pos.y)*.45f);
    }
    if(Vector3.Dot(Vector3.Cross(b-a,c-a),normal)>0) triangles.AddRange(new[]{start,start+1,start+2,start,start+2,start+3});

                if(Mathf.Abs(u)<34&&v>-19&&v<30) weight=Mathf.Max(weight,scar(gate,u,v)*.90f);
            }
            for(int slot=0;slot<old.GetLength(2);slot++) paint[z,x,slot]=old[z,x,slot]*(1-weight);
            paint[z,x,index]+=weight;
        }
        AssetDatabase.CreateAsset(d,terrainPaths[i]);
        d.SetAlphamaps(0,0,paint);edited[i].Flush();AssetDatabase.SaveAssetIfDirty(d);
    }
    Physics.SyncTransforms();
    if(!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene))
        throw new InvalidOperationException("권역 작업장 저장 실패");
    Undo.CollapseUndoOperations(undo);
    return string.Join(" / ",names)+" / 신규 구간 메시 "+meshes.Count;
}
catch
{
    Undo.RevertAllDownToGroup(undo);
    throw;
}
