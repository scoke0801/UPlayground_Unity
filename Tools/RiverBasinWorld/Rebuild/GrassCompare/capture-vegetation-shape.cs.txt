// Unity MCP에서 실행한다. stage는 baseline 또는 shape-wind다. 제작 파일은 읽기 전용으로 연다.
const string stage = "baseline";
// 이름으로 식별하는 두 에디터 저작 프로토타입의 배율을 변경한다.
Func<TerrainData, Vector3, string> shapeDetails = (data, origin) => {
    int width = data.detailWidth, height = data.detailHeight;
    float cellX = data.size.x / width, cellZ = data.size.z / height;
    var existing = data.detailPrototypes;
    var prototypes = existing.ToList();
    var layers = new int[existing.Length][,];
    for (int i = 0; i < existing.Length; i++)
        layers[i] = data.GetDetailLayer(0, 0, width, height, i);

    var updates = new System.Collections.Generic.Dictionary<int, int[,]>();
    var report = new System.Text.StringBuilder();
    for (int index = 0; index < existing.Length; index++)
    {
        var prefab = existing[index].prototype;
        if (!prefab) throw new InvalidOperationException(data.name + " 풀 프리팹 누락");
        bool tall = prefab.name == "PF_FW_TallGrass_02_Autumn_Detail";
        if (!tall && prefab.name != "PF_FW_Silvergrass_03_Detail") continue;
        var shaped = data.detailPrototypes[index];
        float meshHeight = prefab.GetComponent<MeshFilter>().sharedMesh.bounds.size.y;
        shaped.minWidth = tall ? 0.34f : 0.40f;
        shaped.maxWidth = tall ? 0.56f : 0.61f;
        shaped.minHeight = (tall ? 0.50f : 0.56f) / meshHeight;
        shaped.maxHeight = 0.90f / meshHeight;
        var inner = new int[height, width];
        var outer = new int[height, width];
        long inside = 0, outside = 0;
        for (int z = 0; z < height; z++)
            for (int x = 0; x < width; x++)
            {
                int value = layers[index][z, x];
                if (value == 0) continue;
                float wx = origin.x + (x + 0.5f) * cellX;
                float wz = origin.z + (z + 0.5f) * cellZ;
                if (wx >= 440 && wx < 1450 && wz >= 1600 && wz < 2900)
                {
                    inside += value;
                    inner[z, x] = value;
                }
                else
                {
                    outside += value;
                    outer[z, x] = value;
                }
            }

        if (inside == 0) continue;
        if (outside == 0)
            prototypes[index] = shaped;
        else
        {
            updates[index] = outer;
            updates[prototypes.Count] = inner;
            prototypes.Add(shaped);
        }
        report.AppendLine(data.name + " " + prefab.name + " 숲 안 배율 조정 " + inside +
            " / 밖 기존 배율 " + outside + " / 높이 " + (shaped.minHeight * meshHeight) + "–0.90m");
    }

    data.detailPrototypes = prototypes.ToArray();
    for (int i = 0; i < prototypes.Count; i++)
    {
        int[,] expected;
        if (!updates.TryGetValue(i, out expected)) expected = layers[i];
        data.SetDetailLayer(0, 0, i, expected);
        var actual = data.GetDetailLayer(0, 0, width, height, i);
        for (int z = 0; z < height; z++)
            for (int x = 0; x < width; x++)
                if (expected[z, x] != actual[z, x])
                    throw new InvalidOperationException(data.name + " 디테일 값 불일치 " + i + " @ " + x + "," + z);
    }

    return report.ToString();
};

const string source = "Assets/WorldBuilder/Worlds/6effd90b68ea4b3d8f8caa5197fcfe34/Preview/RiverBasin4km.unity";
const string output = "output/RiverBasinWorld/GrassReferencePhase2";
var active = UnityEngine.SceneManagement.SceneManager.GetActiveScene();
if (active.path != source || active.isDirty || EditorApplication.isPlayingOrWillChangePlaymode)
    throw new InvalidOperationException("저장된 제작 씬의 편집 모드에서 실행합니다.");
var preview = UnityEditor.SceneManagement.EditorSceneManager.OpenPreviewScene(source);
var resources = new System.Collections.Generic.List<UnityEngine.Object>();
var oldWind = Shader.GetGlobalVector("_LW_Wind");
var oldPreviousWind = Shader.GetGlobalVector("_LW_PreviousWind");
var oldTarget = RenderTexture.active;
var target = RenderTexture.GetTemporary(1280, 720, 24);
var pixels = new Texture2D(target.width, target.height, TextureFormat.RGB24, false);
int errors = 0;
Application.LogCallback countErrors = (message, stack, type) => {
    if (type == LogType.Error || type == LogType.Exception || type == LogType.Assert) errors++;
};
Application.logMessageReceived += countErrors;
try
{
    System.IO.Directory.CreateDirectory(output);
    var tiles = new System.Collections.Generic.List<Terrain>();
    foreach (var root in preview.GetRootGameObjects()) tiles.AddRange(root.GetComponentsInChildren<Terrain>());
    var names = new[] { "interior", "ridge", "clearing", "low-meadow", "tile-seam", "outside", "shrine-ridge", "upper-west" };
    var poses = new[] {
        new Vector4(1000, 2450, 2.0f, 260),
        new Vector4(1360, 2850, 2.1f, 35),
        new Vector4(1130, 2330, 2.0f, 185),
        new Vector4(1143, 2320, 0.95f, 185),
        new Vector4(1034, 2415, 9, 270),
        new Vector4(1459, 2520, 6, 90),
        new Vector4(1310, 3235, 1.6f, 125),
        new Vector4(940, 3240, 2f, 120)
    };
    var pitch = new[] { 5f, 7f, 6f, 1f, 17f, 15f, 5f, 8f };
    var rootObject = new GameObject("서부 숲 보정 검증 카메라");
    UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(rootObject, preview);
    var camera = rootObject.AddComponent<Camera>();
    camera.enabled = false;
    camera.scene = preview;
    camera.cameraType = CameraType.Game;
    camera.fieldOfView = 54;
    camera.nearClipPlane = 0.1f;
    camera.farClipPlane = 1100;
    camera.targetTexture = target;
    rootObject.AddComponent<UnityEngine.Rendering.Universal.UniversalAdditionalCameraData>().renderPostProcessing = true;

    var notes = new System.Text.StringBuilder();
    if (stage == "shape-wind" || stage == "scoped-shape-wind")
    {
        var sourceShader = AssetDatabase.LoadAssetAtPath<Shader>("Assets/06.Shaders/Foliage/UPG_Grass.shader");
        var materialCopies = new System.Collections.Generic.Dictionary<GameObject, GameObject>();
        int count = 0;
        foreach (var tile in tiles)
        {
            Vector3 origin = tile.GetPosition();
            var original = tile.terrainData;
            bool inScope = origin.x < 1450 && origin.z < 2900 && origin.x + original.size.x > 440 && origin.z + original.size.z > 1600;
            bool sharedWind = original.detailPrototypes.Any(d => d.prototype &&
                d.prototype.name == "PF_FW_TallGrass_02_Autumn_Detail");
            if (!inScope && !sharedWind) continue;

            var copy = UnityEngine.Object.Instantiate(original);
            resources.Add(copy);
            if (inScope) notes.Append(shapeDetails(copy, origin));
            var prototypes = copy.detailPrototypes;
            foreach (var detail in prototypes)
            {
                var prefab = detail.prototype;
                if (!prefab) throw new InvalidOperationException(tile.name + " 풀 프리팹 누락");
                MeshRenderer renderer = prefab.GetComponent<MeshRenderer>();
                if (!renderer || !renderer.sharedMaterial || !renderer.sharedMaterial.shader)
                    throw new InvalidOperationException(prefab.name + " 풀 재질 누락");

                if (renderer.sharedMaterial.shader != sourceShader ||
                    !renderer.sharedMaterial.name.StartsWith("FW_MI_Grass_", StringComparison.Ordinal)) continue;
                GameObject prefabCopy;
                if (!materialCopies.TryGetValue(prefab, out prefabCopy))
                {
                    var material = new Material(renderer.sharedMaterial);
                    resources.Add(material);
                    float bend = material.name.Contains("Tall") ? 0.32f : 0.23f;
                    if (!material.HasProperty("_WindDirectionalBend"))
                        throw new InvalidOperationException("방향성 바람 셰이더를 먼저 가져와야 합니다.");
                    material.SetFloat("_WindDirectionalBend", bend);
                    prefabCopy = (GameObject)PrefabUtility.InstantiatePrefab(prefab, preview);
                    prefabCopy.transform.position = new Vector3(0, -5000, 0);
                    prefabCopy.GetComponent<MeshRenderer>().sharedMaterial = material;
                    materialCopies.Add(prefab, prefabCopy);
                    notes.AppendLine(renderer.sharedMaterial.name + " 방향성 휨 " + bend + "m");
                }
                detail.prototype = prefabCopy;
            }
            copy.detailPrototypes = prototypes;
            tile.terrainData = copy;
            tile.Flush();
            count++;
        }
        if (count == 0 || materialCopies.Count != 7)
            throw new InvalidOperationException("비교용 서부 숲 타일 또는 바람 프로토타입 수가 예상과 다릅니다.");
    }

    // 편집 모드에서 동일 풍향을 재현한다. Game 뷰 검증은 실제 날씨 시간표를 사용한다.
    Shader.SetGlobalVector("_LW_Wind", new Vector4(0.6f, 0f, 0.35f, 0.695f));
    Shader.SetGlobalVector("_LW_PreviousWind", new Vector4(0.6f, 0f, 0.35f, 0.695f));
    for (int shot = 0; shot < poses.Length; shot++)
    {
        var pose = poses[shot];
        var point = new Vector3(pose.x, 0, pose.y);
        var tile = tiles.Find(t => {
            var p = t.GetPosition();
            var size = t.terrainData.size;
            return point.x >= p.x && point.x < p.x + size.x && point.z >= p.z && point.z < p.z + size.z;
        });
        if (!tile) throw new InvalidOperationException("카메라 아래 지형을 찾지 못했습니다. " + names[shot]);
        point.y = tile.SampleHeight(point) + tile.GetPosition().y + pose.z;
        camera.transform.SetPositionAndRotation(point, Quaternion.Euler(pitch[shot], pose.w, 0));
        camera.Render();
        RenderTexture.active = target;
        pixels.ReadPixels(new Rect(0, 0, target.width, target.height), 0, 0);
        pixels.Apply();
        System.IO.File.WriteAllBytes(output + "/" + names[shot] + "-" + stage + ".png", pixels.EncodeToPNG());
    }
    System.IO.File.WriteAllText(output + "/" + stage + "-notes.txt", notes.ToString() + "렌더 오류 " + errors);
    if (errors != 0) throw new InvalidOperationException("보정 확인 캡처에서 Unity 오류 " + errors + "건이 발생했습니다.");
    return stage + " 캡처 " + poses.Length + "장 / Unity 오류 0건";
}
finally
{
    Application.logMessageReceived -= countErrors;
    RenderTexture.active = oldTarget;
    RenderTexture.ReleaseTemporary(target);
    UnityEngine.Object.DestroyImmediate(pixels);
    UnityEditor.SceneManagement.EditorSceneManager.ClosePreviewScene(preview);
    foreach (var resource in resources) UnityEngine.Object.DestroyImmediate(resource);
    Shader.SetGlobalVector("_LW_Wind", oldWind);
    Shader.SetGlobalVector("_LW_PreviousWind", oldPreviousWind);
}
