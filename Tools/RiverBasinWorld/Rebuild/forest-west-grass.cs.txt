System.Func<float, float, float> HeightAt = (x, z) => {
    int tx = Mathf.Clamp(Mathf.FloorToInt(x / 512f), 0, 7), tz = Mathf.Clamp(Mathf.FloorToInt(z / 512f), 0, 7);
    var a = H(tx, tz); float u = (x - tx * 512f), v = (z - tz * 512f);
    int i = Mathf.Clamp((int)u, 0, 511), j = Mathf.Clamp((int)v, 0, 511); float fu = Mathf.Clamp01(u - i), fv = Mathf.Clamp01(v - j);
    return Mathf.Lerp(Mathf.Lerp(a[j, i], a[j, i + 1], fu), Mathf.Lerp(a[j + 1, i], a[j + 1, i + 1], fu), fv) * 800f - 32f;
};

System.Func<float, float, float> RouteDist = (x, z) => {
    float best = 999f;
    foreach (var s in routes) {
        float abx = s.z - s.x, abz = s.w - s.y, apx = x - s.x, apz = z - s.y;
        float tt = Mathf.Clamp01((apx * abx + apz * abz) / (abx * abx + abz * abz));
        float ddx = apx - abx * tt, ddz = apz - abz * tt; float d2 = ddx * ddx + ddz * ddz; if (d2 < best * best) best = Mathf.Sqrt(d2);
    }
    return best;
};
System.Func<float, float, float, float, float, float, float> Ell = (x, z, cx, cz, rx, rz) => Mathf.Sqrt(((x - cx) / rx) * ((x - cx) / rx) + ((z - cz) / rz) * ((z - cz) / rz));
System.Func<float, float, float, float, float, float, float> SegD = (x, z, ax, az, bx, bz) => {
    float abx = bx - ax, abz = bz - az; float tt = Mathf.Clamp01(((x - ax) * abx + (z - az) * abz) / (abx * abx + abz * abz));
    float dx = x - ax - abx * tt, dz = z - az - abz * tt; return Mathf.Sqrt(dx * dx + dz * dz);
};
System.Func<float, float, float> Mask = (x, z) => {
    float n = (Mathf.PerlinNoise(x * 0.011f + 13.1f, z * 0.011f + 7.7f) - 0.5f) * 60f;
    float dx = Mathf.Max(440f - x, x - 1450f), dz = Mathf.Max(1600f - z, z - 2900f);
    float m = Mathf.Clamp01(1f - (Mathf.Max(dx, dz) + n) / 50f);
    float rd = Mathf.Min(SegD(x, z, 1300, 2860, 1360, 3060), SegD(x, z, 1360, 3060, 1280, 3220)) + n * 0.6f;
    m = Mathf.Max(m, Mathf.Clamp01(1f - (rd - 90f) / 50f));
    float kd = Mathf.Min(SegD(x, z, 1520, 3094, 1400, 3170), SegD(x, z, 1400, 3170, 1280, 3256)) + n * 0.5f;
    m = Mathf.Max(m, Mathf.Clamp01(1f - (kd - 45f) / 30f) * Mathf.Clamp01((1515f - x) / 70f));
    return m;
};

System.Func<float, float, float, float, float> Fbm = (x, z, f, o) =>
    (Mathf.PerlinNoise(x * f + o, z * f + o * 0.7f) * 0.55f + Mathf.PerlinNoise(x * f * 2.1f + o * 1.3f, z * f * 2.1f - o) * 0.30f + Mathf.PerlinNoise(x * f * 4.3f - o, z * f * 4.3f + o * 1.9f) * 0.15f);
System.Func<float, float, float, float> Smooth = (a, b, v) => { float t = Mathf.Clamp01((v - a) / (b - a)); return t * t * (3f - 2f * t); };
uint hseed = (uint)SEED;
System.Func<int, int, int, float> Hash = (ix, iz, salt) => { uint h = (uint)ix * 374761393u + (uint)iz * 668265263u + (uint)salt * 2246822519u + hseed; h = (h ^ (h >> 13)) * 1274126177u; h ^= h >> 16; return (h & 0xFFFFFF) / 16777216f; };

System.Func<float[,], float, float, float> Bi = (f, x, z) => {

    return Mathf.Lerp(Mathf.Lerp(f[i, j], f[i + 1, j], fu), Mathf.Lerp(f[i, j + 1], f[i + 1, j + 1], fu), fv);

        float dx = GX0 + i * CC - px, dz = GZ0 + j * CC - pz; float d = Mathf.Sqrt(dx * dx + dz * dz);
        float s = 1f - Smooth(r * 0.45f, R, d); if (s > shade[i, j]) shade[i, j] = s;

        float edge = Smooth(2.25f, 2.25f + 1.2f + en * 3.5f, rd);
        float sl = SlopeAt(x, z); if (sl > 40f) continue;
        int ix = Mathf.FloorToInt(x * 2f), iz = Mathf.FloorToInt(z * 2f);
        float sh = Bi(shade, x, z); sh = Mathf.Clamp01(sh + (Hash(ix >> 3, iz >> 3, 21) - 0.5f) * 0.25f);
        float macro = Bi(fMacro, x, z), lush = Bi(fLush, x, z), wd = WaterDist(x, z) + (en - 0.5f) * 8f;
        float clump = Mathf.PerlinNoise(x * 0.23f + 91.3f, z * 0.23f + 12.7f); // 약 4m 포기
        float u = Hash(ix, iz, 11), u2 = Hash(ix, iz, 12), u3 = Hash(ix, iz, 13);
        float vigor = Mathf.Lerp(0.30f, 1.05f, Smooth(0.30f, 0.70f, macro));
        float tuft = Mathf.Lerp(0.30f, 1.45f, Smooth(0.25f, 0.75f, clump));
        int L0 = -1, cnt = 1, cat = 4; float dens;
        bool ridge = z > 2720f + (en - 0.5f) * 40f;
        if (wd < 10f) {                                     // 물가: 짙은 녹색 위주, 촘촘
            cat = 0; dens = 0.85f * tuft * Mathf.Lerp(1f, 0.55f, sh) * Smooth(1.5f, 4f, wd);
            L0 = u2 < 0.55f ? 4 : (u2 < 0.80f ? 0 : (u2 < 0.92f ? 3 : 1));
        } else if (sh > 0.68f) {                            // 숲 바닥: 포기 단위로 뭉치고 맨땅 얼룩이 섞임
            cat = 1; float bare = Smooth(0.48f, 0.70f, Bi(fBare, x, z));
            dens = Mathf.Lerp(0.80f, 0.34f, Smooth(0.68f, 1.0f, sh)) * (1f - 0.75f * bare) * Mathf.Lerp(0.20f, 1.55f, Smooth(0.30f, 0.72f, clump));
            L0 = u2 < 0.50f ? 5 : (u2 < 0.85f ? 0 : 1);
            // 맨땅 얼룩에는 풀 대신 낙엽 더미를 드문드문(더미 하나가 약 2.5m)
            if (bare > 0.3f && u3 < 0.014f * bare * m) { L0 = 10; cnt = 1; dens = 1f; u = 0f; }
        } else if (lush > 0.53f) {                          // 무성한 풀밭: 키 큰 풀 섞임, 칸당 최대 2
            cat = 2; dens = Mathf.Lerp(0.80f, 1.15f, Smooth(0.53f, 0.72f, lush)) * tuft * (1f - 0.5f * sh);
            L0 = u2 < 0.40f ? 3 : (u2 < 0.70f ? 2 : (u2 < 0.88f ? 1 : 4));
            if (u < dens * 0.30f) cnt = 2;
        } else if (ridge || sl > 18f) {                     // 건조한 능선·양지 사면: 마른 풀, 성김
            cat = 3; dens = 0.55f * vigor * tuft * (1f - 0.7f * sh);
            L0 = u2 < 0.55f ? 6 : (u2 < 0.85f ? 0 : 2);
        } else {                                            // 보통 숲 틈: 지역 생장 세기로 성김~무성
            cat = 4; dens = 0.70f * vigor * tuft * (1f - 0.75f * sh);
            L0 = u2 < 0.55f ? 0 : (u2 < 0.85f ? 1 : 2);
        }
        dens *= m * edge * gScale;
        if (Reserved(x, z)) dens *= 0.5f;
        // 포켓: 억새 군락·들꽃 군락(그늘이 옅은 곳)
        if (sh < 0.45f && wd > 6f) {
            float sv = Bi(fSilver, x, z), fl = Bi(fFlower, x, z);
            if (sv > 0.66f && clump > 0.5f && u3 < 0.10f * Smooth(0.64f, 0.74f, sv) * m) { L0 = 7; cnt = 1; dens = 1f; cat = 5; u = 0f; }
            else if (fl > 0.64f && u3 > 0.80f && u3 < 0.80f + 0.16f * Smooth(0.64f, 0.74f, fl) * m * edge) { L0 = (ridge || cat == 3 || Hash(ix >> 4, iz >> 4, 31) < 0.6f) ? 8 : 9; cnt = 1; dens = 1f; cat = 5; u = 0f; }
        }
        if (u >= dens) continue;
        layers[L0][j, i] = cnt; perLayer[L0] += cnt; perCat[cat] += cnt; cellsTotal++; instTotal += cnt;
    }
    if (mode == "apply") {
        // 숲 경계(GX0..GX1, GZ0..GZ1) 밖 칸은 기존 층에서 같은 프로토타입이면 옮겨 보존
        var oldProtos = d.detailPrototypes; var oldLayers = new int[oldProtos.Length][,];
        for (int L = 0; L < oldProtos.Length; L++) oldLayers[L] = d.GetDetailLayer(0, 0, d.detailWidth, d.detailHeight, L);
        bool partial = o.x < GX0 || o.x + 512f > GX1 || o.z < GZ0 || o.z + 512f > GZ1;
        if (partial && d.detailWidth == 1024) {
            for (int L = 0; L < oldProtos.Length; L++) {
                int nl = -1; for (int k = 0; k < NP; k++) if (dProtos[k].prototype == oldProtos[L].prototype) { nl = k; break; }
                if (nl < 0 && oldProtos[L].prototype != null) { // 이전 판의 B1 풀은 같은 메시의 숲 재질판으로 옮긴다
                    string on = oldProtos[L].prototype.name;
                    if (on == "PF_B1_SmallGrass_01_Detail") nl = 0; else if (on == "PF_B1_SmallGrass_03_Detail") nl = 1; else if (on == "PF_B1_MediumGrass_02_Detail") nl = 2;
                }
                if (nl < 0) continue;
                for (int j = 0; j < 1024; j++) for (int i = 0; i < 1024; i++) { float x = o.x + (i + 0.5f) * 0.5f, z = o.z + (j + 0.5f) * 0.5f; if (x >= GX0 && x < GX1 && z >= GZ0 && z < GZ1) continue; layers[nl][j, i] = oldLayers[L][j, i]; }
            }
        }
        d.SetDetailScatterMode(DetailScatterMode.InstanceCountMode);
        d.detailPrototypes = dProtos; d.SetDetailResolution(1024, 32);
        for (int L = 0; L < NP; L++) d.SetDetailLayer(0, 0, L, layers[L]);
        t.detailObjectDistance = 90f; t.detailObjectDensity = 1f;
        EditorUtility.SetDirty(d); EditorUtility.SetDirty(t);
    }
}
