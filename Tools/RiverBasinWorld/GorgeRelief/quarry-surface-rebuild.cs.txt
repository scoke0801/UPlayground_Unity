Action<Vector3,Vector3,Vector3,Vector3,Vector3,int> quad=(a,b,c,d,normal,sub)=>
{
    int index=vertices.Count;
    foreach(var p in new[]{a,b,c,d})
    {
        var local=p-anchor;
        vertices.Add(local);
        uv.Add(normal.y>.68f?new Vector2(local.x,local.z)*.24f:
            new Vector2(Mathf.Abs(normal.x)>.7f?local.z:local.x,local.y)*.24f);
    }
    if(Vector3.Dot(Vector3.Cross(b-a,c-a),normal)>0) meshIndices[sub].AddRange(new[]{index,index+1,index+2,index,index+2,index+3});
    else meshIndices[sub].AddRange(new[]{index,index+2,index+1,index,index+3,index+2});
};
Action<string,Transform,Material[],bool> publish=(name,parent,mats,collision)=>
{
    var m=new Mesh{name=name+" · "+parent.name};
    m.SetVertices(vertices); m.SetUVs(0,uv); m.subMeshCount=mats.Length;
    for(int i=0;i<mats.Length;i++) m.SetTriangles(meshIndices[i],i);
    m.RecalculateNormals(); m.RecalculateTangents(); m.RecalculateBounds();
    var marker=group(name,parent,anchor);
    marker.gameObject.isStatic=true;
    marker.gameObject.AddComponent<MeshFilter>().sharedMesh=m;
    marker.gameObject.AddComponent<MeshRenderer>().sharedMaterials=mats;
    if(collision) marker.gameObject.AddComponent<MeshCollider>().sharedMesh=m;
    meshAssets.Add(m);
    vertices.Clear(); uv.Clear();
    foreach(var list in meshIndices) list.Clear();
};

    return "절개면 세 단, 인양 광장, 천막 3채, 선별·적재 구역, 운반길 재질, 사면 나무 "+treesAt.Length+"그루";
}
catch
{
    Undo.RevertAllDownToGroup(undo);
    throw;
}
