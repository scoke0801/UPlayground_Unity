Func<Vector3[],Vector3[]> sample=knots=>
{
    var rounded=new List<Vector3>{knots[0]};
    for(int i=1;i<knots.Length-1;i++)
    {
        var previous=knots[i-1]-knots[i];
        var next=knots[i+1]-knots[i];
        float trim=Mathf.Min(6,Mathf.Min(previous.magnitude,next.magnitude)*.32f);
        var a=knots[i]+previous.normalized*trim;
        var b=knots[i]+next.normalized*trim;
        for(int j=0;j<=5;j++)
        {
            float phase=j/5f;
            rounded.Add(Vector3.Lerp(Vector3.Lerp(a,knots[i],phase),Vector3.Lerp(knots[i],b,phase),phase));
        }
    }
    rounded.Add(knots.Last());
    var samples=new List<Vector3>();
    for(int i=1;i<rounded.Count;i++)
    {
        int count=Math.Max(1,Mathf.CeilToInt(Vector3.Distance(rounded[i-1],rounded[i])/2f));
        for(int j=0;j<count;j++) samples.Add(Vector3.Lerp(rounded[i-1],rounded[i],(float)j/count));
    }
    samples.Add(rounded.Last());
    return samples.ToArray();
};

Func<Vector3[],Vector2,Vector2> nearest=(line,point)=>
{
    var result=new Vector2(float.PositiveInfinity,0);
    for(int i=1;i<line.Length;i++)
    {
        var a=flat(line[i-1]);
        var axis=flat(line[i])-a;
        float t=Mathf.Clamp01(Vector2.Dot(point-a,axis)/axis.sqrMagnitude);
        float distance=Vector2.Distance(point,a+axis*t);
        if(distance<result.x) result=new Vector2(distance,Mathf.Lerp(line[i-1].y,line[i].y,t));
    }
    return result;
};

    d.SetHoles(x0,z0,holes);

        publish("운반길에서 갱구로 이어지는 샛길",exterior,road,false);
        var front=mouths[i]-forward*.94f;
        lamp(exterior,front+Vector3.up*2.62f,8.6f,6f);
        foreach(int sign in new[]{-1,1})
        {
            var torch=front+lateral*sign*2.16f+Vector3.up*1.83f;
            place(sconce,exterior,torch,Quaternion.LookRotation(-forward).eulerAngles.y);
        }
        var stock=front+lateral*3.55f-forward*1.35f;
        stock.y=grade(stock);
        place(supplies,exterior,stock,yaw+39);
        var post=front-lateral*2.92f-forward*1.8f;
        post.y=grade(post);
        var pole=GameObject.CreatePrimitive(PrimitiveType.Cube);
        pole.name="갱구 운반 깃대";
        pole.transform.SetParent(exterior);
        pole.transform.position=post+Vector3.up*1.74f;
        pole.transform.localScale=new Vector3(.14f,3.55f,.14f);
        pole.GetComponent<MeshRenderer>().sharedMaterial=beamMaterial;
        UnityEngine.Object.DestroyImmediate(pole.GetComponent<BoxCollider>());
        var d=post+Vector3.up*3.33f;
        quad(d,d+lateral*.95f,d+lateral*.86f+Vector3.down*1.22f-forward*.14f,
            d+lateral*.04f+Vector3.down*1.39f,-forward);
        publish("갱구의 적갈색 깃발",exterior,rimCloth,false);
    }

    var rooms=group("서쪽과 중앙의 지하 집하장",root.transform,anchor);
    var cardinals=new[]{Vector3.forward,Vector3.right,Vector3.back,Vector3.left};
    for(int i=0;i<2;i++)
    {
        var roomCenter=i==0?westHall:mainHall;
        var room=group(i==0?"서쪽 지하 집하장":"중앙 지하 집하장",rooms,roomCenter);
        var x=Vector3.right*6; var z=Vector3.forward*6;
        quad(roomCenter-x-z,roomCenter+x-z,roomCenter+x+z,roomCenter-x+z,Vector3.up);
        publish("Dng_Floor_QuarryMaze",room,gravel,true);
        var top=roomCenter+Vector3.up*4.4f;
        quad(top-x-z,top+x-z,top+x+z,top-x+z,Vector3.down);
        publish("Dng_Roof_QuarryMaze",room,stone,true);
        for(int j=0;j<4;j++)
        {
            var direction=cardinals[j];
            var side=Vector3.Cross(Vector3.up,direction);
            var center=roomCenter+direction*6;
            var l=center-side*6;var r=center+side*6;
            var up=Vector3.up*4.4f;
            if(i==0&&j==2) {quad(l,r,r+up,l+up,-direction);continue;}
            var opening=ring(center,side);
            quad(l,opening[0],opening[1]+Vector3.up*1.75f,l+up,-direction);
            quad(opening[5],r,r+up,opening[4]+Vector3.up*1.75f,-direction);
            for(int k=1;k<4;k++)
                quad(opening[k],opening[k+1],new Vector3(opening[k+1].x,top.y,opening[k+1].z),
                    new Vector3(opening[k].x,top.y,opening[k].z),-direction);
        }
        publish("Dng_Wall_QuarryRegionalRoom",room,stone,true);
        foreach(int sign in new[]{-1,1}) lamp(room,roomCenter+new Vector3(sign*3.1f,3.2f,sign*3.1f),14,4.5f);
        place(supplies,room,roomCenter+new Vector3(4.5f,0,4.5f),-37);
        place(supplies,room,roomCenter+new Vector3(-4.55f,0,-4.55f),142);
    }

    Physics.SyncTransforms();
    var notes=new List<string>();
    int floorCount=0;
    foreach(var line in routes)
    foreach(var q in line.Skip(1))
    {
        RaycastHit hit;
        if(!Physics.Raycast(q+Vector3.up*.85f,Vector3.down,out hit,1.25f,~0,QueryTriggerInteraction.Ignore)||
            Mathf.Abs(hit.point.y-q.y)>.35f) notes.Add(q.ToString("F2"));
        floorCount++;
    }
    if(notes.Count>0) throw new InvalidOperationException("새 갱도 바닥 "+string.Join(", ",notes.Take(12)));
    AssetDatabase.CreateAsset(meshes[0],meshPath);
    foreach(var mesh in meshes.Skip(1)) AssetDatabase.AddObjectToAsset(mesh,meshPath);
    AssetDatabase.AddObjectToAsset(restoredMesh,meshPath);
    AssetDatabase.AddObjectToAsset(rimCloth,meshPath);
    AssetDatabase.SaveAssetIfDirty(meshes[0]);
    for(int i=0;i<variant.Length;i++)
    {
        AssetDatabase.CreateAsset(variant[i],terrainPaths[i]);
        variant[i].SetAlphamaps(0,0,paints[i]);
        editable[i].Flush();
        AssetDatabase.SaveAssetIfDirty(variant[i]);
    }
    if(!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene))
        throw new InvalidOperationException("권역 갱구 씬 저장 실패");
    Undo.CollapseUndoOperations(undo);
    return string.Join(" / ",sites.Select((name,i)=>name+" "+mouths[i]))+" / 갱도 바닥 "+floorCount;
}
catch
{
    Undo.RevertAllDownToGroup(undo);
    throw;
}
