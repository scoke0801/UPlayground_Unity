        Quaternion.Euler((next()-.5f)*tilt,next()*360,(next()-.5f)*tilt));
    var rr=g.GetComponentsInChildren<Renderer>();var b=rr[0].bounds;foreach(var r in rr)b.Encapsulate(r.bounds);
    g.transform.localScale*=size/Mathf.Max(b.size.x,b.size.z);
    b=rr[0].bounds;foreach(var r in rr)b.Encapsulate(r.bounds);
    float bed=ground(p.x,p.y);
    foreach(var d in new[]{new Vector2(-.24f,0),new Vector2(.24f,0),new Vector2(0,-.24f),new Vector2(0,.24f)})
