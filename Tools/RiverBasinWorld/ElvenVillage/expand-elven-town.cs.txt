Func<Vector2,Vector2,Vector2,Vector2> closest=(p,a,b)=>{
    var d=b-a;
    return a+d*Mathf.Clamp01(Vector2.Dot(p-a,d)/d.sqrMagnitude);
};

    }
    return best;
};

            pts.Add(.5f*((2*b)+(-a+c)*t+(2*a-5*b+4*c-d)*t*t+(-a+3*b-3*c+d)*t*t*t));
