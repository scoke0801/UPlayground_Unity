var runs=new float[Line.Length];
for(int i=1;i<runs.Length;i++) runs[i]=runs[i-1]+Vector2.Distance(Line[i-1],Line[i]);
Func<float,Vector2> at=s=>
{
    for(int i=1;i<runs.Length;i++) if(s<=runs[i]) return Vector2.Lerp(Line[i-1],Line[i],Mathf.InverseLerp(runs[i-1],runs[i],s));
    return Line[Line.Length-1];
};
Func<float,Vector2> normal=s=>{var t=(at(s+9)-at(s-9)).normalized;return new Vector2(-t.y,t.x);};
Func<float,float,float,Vector3> xyz=(s,q,y)=>{var p=at(s)+normal(s)*q;return new Vector3(p.x,y,p.y);};
Func<Vector2,Vector2> project=p=>
{
    float best=float.PositiveInfinity; var result=Vector2.zero;
    for(int i=1;i<Line.Length;i++)
    {
        var v=Line[i]-Line[i-1];float t=Mathf.Clamp01(Vector2.Dot(p-Line[i-1],v)/v.sqrMagnitude);
        var off=p-(Line[i-1]+t*v);if(off.sqrMagnitude>=best)continue;
        best=off.sqrMagnitude;
        result=new Vector2(Mathf.Lerp(runs[i-1],runs[i],t),Mathf.Sign(v.x*off.y-v.y*off.x)*off.magnitude);
    }
    return result;
};
