Func<Vector2,Vector2> nearest=p=>{
    float best=float.PositiveInfinity;
    Vector2 result=Vector2.zero;
    foreach(var point in routes){
        float d=(point-p).sqrMagnitude;
        if(d<best){best=d;result=point;}
    }
    return result;
};
Func<Vector2,float> roadDist=p=>Vector2.Distance(p,nearest(p));
Func<Vector2,float> slope=p=>{
    var t=tileAt(p.x,p.y);
    return t.terrainData.GetSteepness((p.x-t.transform.position.x)/512,(p.y-t.transform.position.z)/512);
};

Func<GameObject,Bounds> bounds=o=>{
    var renderers=o.GetComponentsInChildren<MeshRenderer>();
    var b=renderers[0].bounds;
    for(int i=1;i<renderers.Length;i++)b.Encapsulate(renderers[i].bounds);
    return b;
};
Action<GameObject,Vector3,Vector3,float> fit=(instance,p,span,sink)=>{
    var rotation=instance.transform.rotation;
    instance.transform.rotation=Quaternion.identity;
    var b=bounds(instance);
    instance.transform.localScale=Vector3.Scale(instance.transform.localScale,new Vector3(span.x/b.size.x,span.y/b.size.y,span.z/b.size.z));
    instance.transform.rotation=rotation;
    b=bounds(instance);
    instance.transform.position+=new Vector3(p.x-b.center.x,p.y-b.min.y-sink,p.z-b.center.z);
};

    for(int k=1;k<meshes.Count;k++)AssetDatabase.AddObjectToAsset(meshes[k],atlasPath);
    AssetDatabase.AddObjectToAsset(trailMat,atlasPath);
    AssetDatabase.SaveAssetIfDirty(meshes[0]);
    foreach(var tile in paintBackups.Keys)AssetDatabase.SaveAssetIfDirty(tile.terrainData);
    if(!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene))throw new InvalidOperationException("채석장 프리뷰 저장 실패");
    Undo.CollapseUndoOperations(undo);
    return "장소 "+centers.Length+" 암반 "+rocks+" 판석 "+slabs+" 채광 "+mines+" 샛길 "+tracks+"\n"+report.ToString();
}
catch
{
    foreach(var item in paintBackups)item.Key.terrainData.SetAlphamaps(0,0,item.Value);
    Undo.RevertAllDownToGroup(undo);
    throw;
}
