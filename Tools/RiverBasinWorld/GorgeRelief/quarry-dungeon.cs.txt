Func<Vector3, Vector3, Vector2> project = (p,n) =>
{
    if (Mathf.Abs(n.y) > .56f) return new Vector2(p.x,p.z) * .57f;
    return new Vector2(Mathf.Abs(n.x) > Mathf.Abs(n.z) ? p.z : p.x,p.y) * .57f;
};
Action<Vector3, Vector3, Vector3, Vector3> tri = (a,b,c,toward) =>
{
    if (Vector3.Dot(Vector3.Cross(b-a,c-a),toward) < 0) { var swap=b; b=c; c=swap; }
    foreach (var p in new[] {a,b,c})
    {
        indices.Add(v.Count);
        v.Add(p-origin);
        uv.Add(project(p-origin,toward));
    }
};
Action<Vector3, Vector3, Vector3, Vector3, Vector3> quad = (a,b,c,d,n) =>
{
    tri(a,b,c,n);
    tri(a,c,d,n);
};
Func<string,Mesh> finishMesh = name =>
{
    var mesh = new Mesh { name = name };
    mesh.SetVertices(v);
    mesh.SetUVs(0,uv);
    mesh.SetTriangles(indices,0);
    mesh.RecalculateNormals();
    mesh.RecalculateTangents();
    mesh.RecalculateBounds();
    v.Clear();
    uv.Clear();
    indices.Clear();
    meshAssets.Add(mesh);
    return mesh;
};

{
    float x=p.x, z=p.y;
    float radius = Mathf.Sqrt(Mathf.Pow((x-3201)/22f,2)+Mathf.Pow((z-2204)/26f,2));
    float edge = radius + (Mathf.PerlinNoise(x*.24f,z*.24f)-.5f)*.29f;
    if (x>3214.1f || edge>1.12f || random()>Mathf.Clamp01((1.12f-edge)*4.5f)) continue;

    var polygon = new List<Vector2> {p+new Vector2(-1,-1),p+new Vector2(1,-1),p+new Vector2(1,1),p+new Vector2(-1,1)};
    foreach (var q in sites.Where(q => (q-p).sqrMagnitude < 7 && q!=p))
    {
        var clipped = new List<Vector2>();
        var mid = (p+q)*.5f;
        var axis = q-p;
        for (int i=0;i<polygon.Count;i++)
        {
            var a=polygon[i];
            var b=polygon[(i+1)%polygon.Count];
            float da=Vector2.Dot(a-mid,axis), db=Vector2.Dot(b-mid,axis);
            if (da<=0) clipped.Add(a);
            if ((da<=0)!=(db<=0)) clipped.Add(Vector2.Lerp(a,b,da/(da-db)));
        }
        polygon=clipped;
    }
    float size = Mathf.Lerp(.65f,.96f,Mathf.Clamp01((1.15f-edge)*2.9f));
    float height = Mathf.Lerp(.035f,.07f,random());
    var center = new Vector3(x,ground(x,z)+height,z);
    for (int i=0;i<polygon.Count;i++)
    {
        var aa=Vector2.Lerp(p,polygon[i],size);
        var bb=Vector2.Lerp(p,polygon[(i+1)%polygon.Count],size);
        var a=new Vector3(aa.x,ground(aa.x,aa.y)+height,aa.y);
        var b=new Vector3(bb.x,ground(bb.x,bb.y)+height,bb.y);
        tri(center,a,b,Vector3.up);
        var side=(a+b)*.5f-center;
        side.y=0;
        quad(a,b,b-Vector3.up*(height+.025f),a-Vector3.up*(height+.025f),side);
    }
    plates++;
}

        objects.Single(x=>x.name=="묘실 촛대 빛" && x.parent.name=="RiverBasin · 처마와 생활 조명")})
        move(target,target.position+Vector3.up*rise);
    deactivate(corridor.Find("폐갱도로 들어가기").gameObject);
    deactivate(corridor.Find("채석장 아래로 돌아가기").gameObject);
    var buttress=crypt.Cast<Transform>().Single(x=>x.name=="Dng_Buttress" && x.position.x>3211);
    move(buttress,new Vector3(3210.15f,cryptFloor,2209.2f));
    foreach(var b in crypt.Cast<Transform>().Where(x=>x.name=="Dng_Brazier" && x.position.x<3200))
    {
        var p=b.position;
        p.z=2204+Mathf.Sign(p.z-2204)*1.66f;
        move(b,p);
    }

    Func<GameObject,Transform,Vector3,Quaternion,GameObject> place = (prefab,parent,p,q) =>
    {
        var g=(GameObject)PrefabUtility.InstantiatePrefab(prefab,parent);
        Undo.RegisterCreatedObjectUndo(g,"갱도 소품");
        g.transform.SetPositionAndRotation(p,q);
        return g;
    };
    var innerArch=crypt.Cast<Transform>().Single(x=>x.name=="Dng_Arch" && x.position.x>3210);
    var header=crypt.Cast<Transform>().Single(x=>x.name=="Dng_Wall_Plain"
        && Mathf.Abs(x.position.x-3214)<.02f && x.position.y>74 && Mathf.Abs(x.position.z-2204)<.01f);
    foreach(var source in new[] { innerArch,header })
    {
        var g=place(PrefabUtility.GetCorrespondingObjectFromSource(source.gameObject),crypt,
            new Vector3(3212,source.position.y,2210.02f),Quaternion.identity);
        var scale=source.localScale;
        if(source==innerArch) scale.x*=1.18f;
        g.transform.localScale=scale;
        var renderer=g.GetComponentInChildren<MeshRenderer>();
        renderer.sharedMaterials=source.GetComponentInChildren<MeshRenderer>().sharedMaterials;
        PrefabUtility.RecordPrefabInstancePropertyModifications(renderer);
    }
    var fixturePositions = new[] {
        new Vector4(3218,94.65f,2201.3f,0),new Vector4(3225,91.22f,2206.7f,180),
        new Vector4(3231,85.8f,2208.7f,90),new Vector4(3235,85.14f,2213.7f,180),
        new Vector4(3241.66f,86.18f,2208,270),new Vector4(3230.2f,82.64f,2221.4f,90),
        new Vector4(3224.3f,80.68f,2225.67f,180),new Vector4(3223,80.54f,2214.32f,0),
        new Vector4(3214.4f,77.14f,2217.66f,180),new Vector4(3213.66f,75.14f,2212.4f,270) };
    foreach (var p in fixturePositions)
    {
        var pos=new Vector3(p.x,p.y,p.z);
        var rotation=Quaternion.Euler(0,p.w,0);
        place(sconce,maze,pos,rotation);
        point("갱도 등잔빛",maze,pos+rotation*Vector3.forward*.42f+Vector3.up*.2f,7.6f,1.9f,amber);
    }
    // 목재 지보는 바닥 중앙과 세 갈래 출입 폭을 비워 둔다.
    foreach (var p in new[]{new Vector3(3220,91.57f,2204),new Vector3(3226,88.75f,2204),new Vector3(3232,84.36f,2209.1f),new Vector3(3224,78.98f,2216)})
    {
        var section=make("목재 지보",maze,p).transform;
        bool hub=p.x>3230;
        float half=p.x<3230 && p.z==2204?2.72f:1.73f;
        var sectionRotation=Quaternion.Euler(0,hub?90:0,0);
        var offset=hub?Vector3.right:Vector3.forward;
        for (int i=0;i<3;i++)
        {
            bool lintel=i==2;
            var center=p+(lintel?Vector3.up*3.43f:Vector3.up*1.77f+(i==0?-1:1)*half*offset);
            var part=place(beam,section,p,sectionRotation*(lintel?Quaternion.Euler(0,90,0):Quaternion.Euler(0,0,90)));
            if(lintel) part.transform.localScale=new Vector3((half*2+.3f)/3.7f,1,1);
            part.transform.position+=center-part.GetComponentInChildren<MeshRenderer>().bounds.center;
        }
    }
    var sockets = new[] {new Vector4(3240,82.4f,2217.95f,0),new Vector4(3237.95f,79.6f,2224,90),new Vector4(3206.05f,72.8f,2212,90)};
    for(int i=0;i<sockets.Length;i++)
    {
        var socket=sockets[i];
        var p=new Vector3(socket.x,socket.y,socket.z);
        var g=make("미굴착 접속부 "+(i+1),maze,p);
        g.transform.rotation=Quaternion.Euler(0,socket.w,0);
        attach("Dng_Wall_QuarryExpansionSeal",g.transform,sealMeshes[i],cavern,true);
        foreach(int sign in new[]{-1,1})
        {
            var strut=place(beam,g.transform,p+g.transform.right*sign*.87f+Vector3.up*1.84f,
                g.transform.rotation*Quaternion.Euler(0,0,sign*36));
            strut.name="폐갱도 교차 버팀목";
            var center=p+Vector3.up*1.95f+g.transform.right*sign*.66f;
            strut.transform.position+=center-strut.GetComponentInChildren<MeshRenderer>().bounds.center;
        }
    }

    terrainCopy=UnityEngine.Object.Instantiate(sourceTerrain);
    var map=sourceTerrain.GetAlphamaps(0,0,sourceTerrain.alphamapWidth,sourceTerrain.alphamapHeight);
    var rockIndex=Array.FindIndex(sourceTerrain.terrainLayers,x=>x.name=="Rock");
    var weatherIndex=Array.FindIndex(sourceTerrain.terrainLayers,x=>x.name=="WeatheredRock");
    var shoulders=new[] { new Vector4(3198,2223,33,22),new Vector4(3178,2205,28,22),new Vector4(3226,2217,19,27) };
    for(int z=0;z<sourceTerrain.alphamapHeight;z++)
    for(int x=0;x<sourceTerrain.alphamapWidth;x++)
    {
        float gx=terrain.transform.position.x+(x+.5f)*terrainCopy.size.x/sourceTerrain.alphamapWidth;
        float gz=terrain.transform.position.z+(z+.5f)*terrainCopy.size.z/sourceTerrain.alphamapHeight;
        float distance=shoulders.Min(c=>Mathf.Sqrt(Mathf.Pow((gx-c.x)/c.z,2)+Mathf.Pow((gz-c.y)/c.w,2)));
        float noise=(Mathf.PerlinNoise(gx*.072f,gz*.101f)-.5f)*.35f;
        float weight=Mathf.SmoothStep(0,.73f,Mathf.Clamp01((1.02f-distance+noise)*1.5f));
        if(weight<.025f) continue;
        for(int layer=0;layer<sourceTerrain.alphamapLayers;layer++) map[z,x,layer]*=1-weight;
        map[z,x,rockIndex]+=weight*.69f;
        map[z,x,weatherIndex]+=weight*.31f;
    }
    terrainCopy.SetAlphamaps(0,0,map);
    var patches=terrainCopy.GetHoles(3215-3072,2201-2048,13,6);
    for(int z=0;z<patches.GetLength(0);z++)
    for(int x=0;x<patches.GetLength(1);x++) patches[z,x]=false;
    terrainCopy.SetHoles(3215-3072,2201-2048,patches);
    var tc=terrain.GetComponent<TerrainCollider>();
    Undo.RecordObjects(new UnityEngine.Object[]{terrain,tc},"갱도 입구 지형 구멍");
    terrain.terrainData=terrainCopy;
    tc.terrainData=terrainCopy;
    terrain.Flush();
    Physics.SyncTransforms();

    if (plates<300 || maze.childCount<20 || flame.GetComponentsInChildren<ParticleSystem>().Length==0)
        throw new InvalidOperationException("채석장 저작 구조가 불완전합니다.");
    AssetDatabase.CreateAsset(floors,meshPath);
    foreach(var mesh in meshAssets.Where(x=>x!=floors)) AssetDatabase.AddObjectToAsset(mesh,floors);
    AssetDatabase.AddObjectToAsset(stone,floors);
    AssetDatabase.AddObjectToAsset(cavern,floors);
    AssetDatabase.AddObjectToAsset(entranceRock,floors);
    AssetDatabase.CreateAsset(terrainCopy,terrainPath);
    AssetDatabase.SaveAssetIfDirty(floors);
    AssetDatabase.SaveAssetIfDirty(terrainCopy);
    Undo.FlushUndoRecordObjects();
    UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
    if (!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene))
        throw new InvalidOperationException("채석장 씬 저장에 실패했습니다.");
    Undo.CollapseUndoOperations(transaction);
    return plates+" 판석, "+cells.Count+" 갱도 모듈, "+sockets.Length+" 확장 접속부";
}
catch
{
    Undo.RevertAllDownToGroup(transaction);
    throw;
}
