var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();
var root=scene.GetRootGameObjects().Single(g=>g.name=="채석장 · 첫 시안 배치");
var terrain=Terrain.activeTerrains.Single(t=>t.gameObject.scene==scene&&t.name=="Terrain_6_4");
var terrainData=terrain.terrainData;
var terrainPath=AssetDatabase.GetAssetPath(terrainData);
var oldData=AssetDatabase.LoadAssetAtPath<TerrainData>(terrainPath.Replace("QuarryConcept","QuarryStoneFinish"));
var sceneObjects=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();
Func<string,Transform> get=name=>sceneObjects.Single(t=>t.name==name);
var gates=new[]{get("A 하층 운반 갱구"),get("B 하단 작업장 갱구"),get("C 상단 적재장 갱구"),get("D 강가 보물 동굴")};
var heightOk=gates[0].position.y<gates[1].position.y&&gates[1].position.y<gates[2].position.y;
var seals=Enumerable.Range(1,3).Count(i=>!get("미굴착 접속부 "+i).gameObject.activeInHierarchy);
var walls=root.GetComponentsInChildren<MeshCollider>().Where(c=>c.name=="Dng_Wall_QuarryStoneTerrace").ToArray();
var reward=get("RB_QuarryRiverCave_Treasure");
var report=new System.Text.StringBuilder();
report.AppendLine("고도 A < B < C = "+heightOk+" / 폐쇄면 3곳 개방 "+seals+"/3");
report.AppendLine(string.Join(" / ",gates.Select(t=>t.name+" "+t.position.ToString("F2"))));
report.AppendLine("D 독립 동굴의 기존 보상 구성 = "+reward.gameObject.activeInHierarchy+" / "+
    string.Join(", ",reward.GetComponents<Component>().Where(c=>c).Select(c=>c.GetType().Name)));
report.AppendLine("절개면 충돌과 렌더 메시 일치 "+walls.Count(c=>c.sharedMesh==c.GetComponent<MeshFilter>().sharedMesh)+"/3");
report.AppendLine("지형 렌더와 충돌 일치 "+(terrainData==terrain.GetComponent<TerrainCollider>().terrainData));
report.AppendLine("누락 스크립트 "+root.GetComponentsInChildren<Transform>(true).Sum(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)));
report.AppendLine("누락 또는 오류 재질 "+root.GetComponentsInChildren<Renderer>().Count(r=>r.sharedMaterials.Any(m=>!m||!m.shader||m.shader.name=="Hidden/InternalErrorShader")));

var origin=terrain.transform.position;
var alpha=terrainData.GetAlphamaps(0,0,terrainData.alphamapWidth,terrainData.alphamapHeight);
var oldAlpha=oldData.GetAlphamaps(0,0,oldData.alphamapWidth,oldData.alphamapHeight);
int paintOutside=0;
for(int z=0;z<alpha.GetLength(0);z++)
for(int x=0;x<alpha.GetLength(1);x++)
{
    float wx=origin.x+(x+.5f)*512/alpha.GetLength(1),wz=origin.z+(z+.5f)*512/alpha.GetLength(0);
    if (wx>=3198&&wx<=3254&&wz>=2228&&wz<=2270) continue;
    for(int layer=0;layer<alpha.GetLength(2);layer++)
        if(Mathf.Abs(alpha[z,x,layer]-oldAlpha[z,x,layer])>.004f) paintOutside++;
}
report.AppendLine("작업면 도색 범위 밖 변경 "+paintOutside);

// 선언된 경로의 중심과 좌우를 물리 바닥과 보행 캡슐로 표본 검사한다.
Func<float,float,float,Vector3> p=(x,y,z)=>new Vector3(x,y,z);
var routes=new Dictionary<string,Vector3[]> {
    {"강변 운반길 → A",new[]{p(3180,74,2253),p(3176,73.25f,2249),p(3172,73.1f,2243),
        p(3170,74.25f,2237),p(3169,75.6f,2231),p(3168.7f,75.55f,2226),p(3169,75.65f,2222.5f),p(3171.5f,76.05f,2220),gates[0].position}},
    {"A → 묘실",new[]{gates[0].position,p(3180,75.85f,2220),p(3187,74.9f,2219.4f),p(3194,74.15f,2216.5f),
        p(3200,73.2f,2212.1f),p(3203,72.8f,2212),p(3208,72.8f,2212),p(3212,72.9f,2212),p(3212,71.5f,2208),p(3212,71.5f,2205)}},
    {"B → 하부 순환로",new[]{gates[1].position,p(3253,79.1f,2259),p(3247,79,2251),p(3248,79.2f,2233),
        p(3247,79.6f,2224),p(3237.8f,79.6f,2224),p(3232,79.8f,2224),p(3232,81,2220),p(3232,81.9f,2216),
        p(3228,80.6f,2216),p(3224,79,2216),p(3220,77.1f,2216),p(3216,75.7f,2216),p(3216,76,2220),p(3216,76.7f,2224),
        p(3220,77.6f,2224),p(3224,78.6f,2224),p(3228,79.4f,2224),p(3232,79.8f,2224)}},
    {"C → 상부 순환로",new[]{gates[2].position,p(3252,86.7f,2200),p(3247,85.8f,2201),p(3247,84.2f,2212),
        p(3246,83.6f,2222),p(3240,83.65f,2223),p(3240,82.4f,2217.8f),p(3240,82.4f,2216),p(3240,83.2f,2212),
        p(3240,84.6f,2208),p(3240,85.5f,2204),p(3236,86,2204),p(3232,86,2204),p(3232,84.7f,2208),
        p(3232,83.2f,2212),p(3236,83.2f,2212),p(3240,83.2f,2212)}},
    {"상하부 순환로 → 묘실",new[]{p(3232,83.2f,2212),p(3232,81.9f,2216),p(3228,80.6f,2216),p(3224,79,2216),
        p(3220,77.1f,2216),p(3216,75.7f,2216),p(3212,74.4f,2216),p(3212,72.9f,2212),p(3212,71.5f,2208)}},
    {"D 진입 → 상자 앞",new[]{gates[3].position,p(3149,66.4f,2248),p(3154,66.2f,2249),p(3159,66.2f,2253),p(3161.2f,66.2f,2253)}},
    {"지상 기존 순환 운반길",new[]{p(3188,94,2207),p(3184,88,2220),p(3184,80,2240),p(3196,77,2257),p(3214,77,2264),
        p(3240,76,2271),p(3255,78,2272),p(3268,81,2264),p(3269,85,2234),p(3268,89,2216),p(3266,90,2200),
        p(3264,90,2190),p(3255,97,2178),p(3241,99,2165),p(3218,99,2166),p(3198,99,2176),p(3189,97,2191),p(3188,94,2207)}}
};
Physics.SyncTransforms();
foreach(var pair in routes)
{
    int count=0,floorMiss=0,blocked=0;
    var examples=new List<string>();
    var line=pair.Value;
    bool followSurface=pair.Key=="지상 기존 순환 운반길";
    float steepest=0;
    for(int i=1;i<line.Length;i++)
    {
        var a=line[i-1]; var b=line[i];
        var sideline=Vector3.Cross(Vector3.up,b-a).normalized;
        int steps=Mathf.CeilToInt(Vector3.Distance(a,b)/.6f);
        for(int j=0;j<steps;j++)
        foreach(float side in new[]{-.55f,0,.55f})
        {
            count++;
            var q=Vector3.Lerp(a,b,(j+.5f)/steps)+sideline*side;
            if(followSurface) q.y=terrain.SampleHeight(q)+origin.y;
            RaycastHit hit;
            if(!Physics.Raycast(q+Vector3.up*1.3f,Vector3.down,out hit,2.8f,~0,QueryTriggerInteraction.Ignore))
            {
                floorMiss++;
                if(examples.Count<5) examples.Add("바닥 "+q.ToString("F2"));
                continue;
            }
            steepest=Mathf.Max(steepest,Vector3.Angle(hit.normal,Vector3.up));
            var overlap=Physics.OverlapCapsule(hit.point+Vector3.up*.47f,hit.point+Vector3.up*1.43f,.32f,~0,QueryTriggerInteraction.Ignore);
            var obstacle=overlap.FirstOrDefault(c=>c!=hit.collider);
            if(!obstacle) continue;
            blocked++;
            if(examples.Count<5) examples.Add(obstacle.name+" "+q.ToString("F2"));
        }
    }
    report.AppendLine(pair.Key+" 표본 "+count+" 바닥 누락 "+floorMiss+" 겹침 "+blocked+" 최대 경사 "+steepest.ToString("F1"));
    foreach(var example in examples) report.AppendLine("  "+example);
}
return report.ToString();
