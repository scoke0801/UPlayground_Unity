Func<Vector2[],Vector2[]> sampleRoute = points => {
    var samples = new System.Collections.Generic.List<Vector2>();
    for (int i = 0; i < points.Length - 1; i++) {
        var a = points[Mathf.Max(0,i-1)];
        var b = points[i];
        var c = points[i+1];
        var d = points[Mathf.Min(points.Length-1,i+2)];
        int steps = Mathf.CeilToInt(Vector2.Distance(b,c)/1.8f);
        for (int j=0; j<steps; j++) {
            float u = j/(float)steps;
            samples.Add(.5f*((2*b)+(-a+c)*u+(2*a-5*b+4*c-d)*u*u+(-a+3*b-3*c+d)*u*u*u));
        }
    }
    samples.Add(points.Last());
    return samples.ToArray();
};

};

Func<Vector2,int,float> relief = (p,i) => {
    var f = deposit[i];
    float angle = (i==4 || i==5 ? 75 : i==3 ? 22 : i==1 ? 48 : -33)*Mathf.Deg2Rad;
    var d = p-new Vector2(f.x,f.y);
    float along = (d.x*Mathf.Cos(angle)+d.y*Mathf.Sin(angle))/f.z;
    float across = -d.x*Mathf.Sin(angle)+d.y*Mathf.Cos(angle)-12*along*along+4*Mathf.Sin(5*along+i);
    float asym = across > 0 ? 26 : 15;
    float noise = Mathf.PerlinNoise(p.x*.068f+i,p.y*.065f)-.5f;
    float cap = 1-along*along-Mathf.Pow(across/asym,2)+noise*.18f;
    return f.w*Mathf.Pow(Mathf.Clamp01(cap),.82f);
};
