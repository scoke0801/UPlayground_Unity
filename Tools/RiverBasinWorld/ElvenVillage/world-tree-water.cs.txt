Func<int,int,Mesh> ribbon=(start,end)=>{
    const int cols=16;
    var v=new List<Vector3>();var uv=new List<Vector2>();var tris=new List<int>();
    for(int j=start;j<=end;j++){
        var row=rows[j];var p=xy(row);
        var d=(xy(rows[Mathf.Min(end,j+1)])-xy(rows[Mathf.Max(start,j-1)])).normalized;
        var side=new Vector2(-d.y,d.x);
        for(int k=0;k<=cols;k++){
            float s=2*k/(float)cols-1;
            var q=p+side*(row.w*(1.02f+(Mathf.PerlinNoise(p.x*.067f+s,p.y*.063f)-.5f)*.019f)*s);
            v.Add(new Vector3(q.x,row.z,q.y));uv.Add(new Vector2(q.x/6,q.y/6));
            if(j==start||k==0)continue;
            int b=v.Count-1,a=b-cols-1;
            tris.AddRange(new[]{a-1,a,b-1,a,b,b-1});
        }
    }
    var mesh=new Mesh();mesh.SetVertices(v);mesh.SetUVs(0,uv);mesh.SetTriangles(tris,0);
    mesh.RecalculateNormals();
    if(mesh.normals[cols/2].y<0){tris.Reverse();mesh.SetTriangles(tris,0);mesh.RecalculateNormals();}
    mesh.RecalculateTangents();mesh.RecalculateBounds();return mesh;
};
