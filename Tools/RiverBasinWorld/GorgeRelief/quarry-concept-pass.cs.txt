Func<Vector3[],Vector2,Vector2> nearest = (line,point) =>
{
    var sample = new Vector2(float.PositiveInfinity,0);
    for (int i=1; i<line.Length; i++)
    {
        var start = new Vector2(line[i-1].x,line[i-1].z);
        var delta = new Vector2(line[i].x-start.x,line[i].z-start.y);
        float along = Mathf.Clamp01(Vector2.Dot(point-start,delta)/delta.sqrMagnitude);
        float distance = Vector2.Distance(point,start+delta*along);
        if (distance<sample.x) sample = new Vector2(distance,Mathf.Lerp(line[i-1].y,line[i].y,along));
    }
    return sample;
};
Func<float,float,float> ground = (x,z) => work.GetInterpolatedHeight((x-origin.x)/work.size.x,(z-origin.z)/work.size.z)+origin.y;

for (int x=0;x<width;x++)
{
    var point=new Vector2(origin.x+startX+x,origin.z+startZ+z);
    var profile=nearest(approach,point);
    float edge=1-Mathf.SmoothStep(0,1,Mathf.InverseLerp(1.55f,5.8f,profile.x));
    float preserveSpur=Mathf.SmoothStep(0,1,Mathf.InverseLerp(0,5,Vector2.Distance(point,new Vector2(3180,2253))));
    float baseHeight=origin.y+heights[z,x]*work.size.y;
    heights[z,x]=(Mathf.Lerp(baseHeight,profile.y,edge*preserveSpur)-origin.y)/work.size.y;
}
work.SetHeights(startX,startZ,heights);

Func<string,Transform,Vector3,GameObject> make=(name,parent,position)=>
{
    var go=new GameObject(name);
    go.transform.SetParent(parent);
    go.transform.position=position;
    return go;
};
Func<GameObject,Transform,Vector3,float,GameObject> prop=(prefab,parent,position,yaw)=>
{
    var go=(GameObject)PrefabUtility.InstantiatePrefab(prefab,parent);
    go.transform.SetPositionAndRotation(position,Quaternion.Euler(0,yaw,0));
    return go;
};
var vertices=new List<Vector3>();
var triangles=new List<int>();
var uvs=new List<Vector2>();
Action<Vector3,Vector3,Vector3,Vector3,Vector3> quad=(a,b,c,d,normal)=>
{
    int index=vertices.Count;
    foreach (var point in new[]{a,b,c,d})
    {
        var local=point-anchor;
        vertices.Add(local);
        if (Mathf.Abs(normal.y)>.7f) uvs.Add(new Vector2(local.x,local.z)*.42f);
        else uvs.Add(new Vector2(Mathf.Abs(normal.x)>.7f?local.z:local.x,local.y)*.42f);
    }
    if (Vector3.Dot(Vector3.Cross(b-a,c-a),normal)>0) triangles.AddRange(new[]{index,index+1,index+2,index,index+2,index+3});
    else triangles.AddRange(new[]{index,index+2,index+1,index,index+3,index+2});
};
Action<string,Transform,Material,bool> mesh=(name,parent,mat,collision)=>
{
    var asset=new Mesh {name=name+" · "+parent.name};
    asset.SetVertices(vertices);
    asset.SetUVs(0,uvs);
    asset.SetTriangles(triangles,0);
    asset.RecalculateNormals();
    asset.RecalculateTangents();
    var go=make(name,parent,anchor);
    go.isStatic=true;
    go.AddComponent<MeshFilter>().sharedMesh=asset;
    go.AddComponent<MeshRenderer>().sharedMaterial=mat;
    if (collision) go.AddComponent<MeshCollider>().sharedMesh=asset;
    meshes.Add(asset);
    vertices.Clear(); triangles.Clear(); uvs.Clear();
};
Action<string,Transform,Vector3,Vector3,Vector3,Material,bool> block=(name,parent,at,size,angles,mat,collision)=>
{
    var go=GameObject.CreatePrimitive(PrimitiveType.Cube);
    go.name=name;
    go.transform.SetParent(parent);
    go.transform.SetPositionAndRotation(at,Quaternion.Euler(angles));
    go.transform.localScale=size;
    go.GetComponent<MeshRenderer>().sharedMaterial=mat;
    if (!collision) UnityEngine.Object.DestroyImmediate(go.GetComponent<BoxCollider>());
};
